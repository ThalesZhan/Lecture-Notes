\chapter{Riemannian Langevin Dynamics}

Let $(M,g)$ be a complete Riemannian manifold with Levi-Civita connection $\nabla$ and the canonical Riemannian volume measure $dm$. In the following, we denote $\op{grad} f = \nabla f$ for convenience. Let $B^M$ is a Brownian motion on $M$. Let $U \in C^\infty(M)$. Consider the following SDE
\begin{equation}\label{eq:RLD}
    dX_t = -\nabla U(X_t) dt + \sqrt{2}dB^M_t,
\end{equation}
which is called a Riemannian Langevin dynamics.

\section{Diffusion Operator}

Because $B^M$ is a Brownian motion on $M$, its anti-development $W$ is a standard Euclidean Brownian motion. So
\begin{equation*}
    dB^M_t = V_\alpha \circ dW^\alpha_t,
\end{equation*}
with
\begin{equation*}
    \Delta_M = \sum_\alpha V_\alpha^2.
\end{equation*}
Therefore,
\begin{equation*}
    dX_t = -\nabla U(X_t) dt + \sqrt{2}V_\alpha \circ dW^\alpha_t,
\end{equation*}
which implies that the diffusion operator of $X$ is
\begin{align*}
    Lf &= \frac{1}{2} \sum_\alpha (\sqrt{2}V_\alpha)^2 f - \nabla U (f) \\
    &= \Delta_M f - \inn{\nabla U,\nabla f},
\end{align*}
i.e., $L = \Delta_M - \inn{\nabla U, \nabla \cdot}$. For such $L$, in note \emph{Stochastic Analysis}, we have shown that 
\begin{equation*}
    L^*p = \Delta_M p + \op{div}(p\nabla U)
\end{equation*}
in the sense of on $L^2(dm)$. Also,
\begin{equation*}
    d\mu = e^{-U}dm,
\end{equation*}
is the invariant measure of such diffusion process, which is also symmetric for $L$. Note that $L$ is essentially self-adjoint on $C_c(M)$ w.s.t. $\mu$. Therefore, it has a unique self-adjoint extension $L$ on $L^2(\mu,M)$. Let
\begin{equation*}
    P_t = e^{-tL}
\end{equation*}
be the functional calculus of $L$, defined on $L^2(\mu,M)$. On the other hand, define
\begin{equation*}
    Q_t f(x) = \E\bj{f(X_t) \mid X_0 = x},\quad \forall~ f \in C_c^\infty(M),
\end{equation*} 
By note \emph{Markov Operator}, $Q_t$ can be extended to $L^2(\mu)$ and has the generator $A$ that is self-adjoint on $L^2(\mu)$. By Dynkin's formula,
\begin{equation*}
    Af = Lf,\quad \forall~f \in C_c^\infty(M).
\end{equation*}
Therefore, $L \subset A$. By the essential self-adjointness of $L$, $L = A$, which implies that
\begin{equation*}
    P_tf(x) = Q_tf(x) = \E\bj{f(X_t) \mid X_0 = x},\quad \forall~f \in L^2(\mu).
\end{equation*}
\begin{rmk}
    Note that if $\nabla U$ is globally Lipschitz continuous (in particular, $M$ is compact), i.e., $\abs{\nabla^2 U} \leq L$, then it can show that $(Q_t)_{t \geq 0}$ is a Feller-Dynkin semigroup so that it can be defined on $C_0(M)$.
\end{rmk}
\begin{rmk}
    For $P_t$, note that
    \begin{equation*}
        \frac{\partial}{\partial t}P_t = LP_t = P_tL,
    \end{equation*}
    called the backward Kolmogorov equation. Moreover, it can define the Markov transition kernel
    \begin{equation*}
        p_t(x,A) = P_t\mathbb{I}_A(x) = \E\bj{\mathbb{I}_A(X_t) \mid X_0 = x} = \Pb_x(X_t \in A).
    \end{equation*}
    Then
    \begin{equation*}
        P_tf(x) = \int_M f(y)p_t(x,dy).
    \end{equation*}
    It follows that we can define
    \begin{equation*}
        \int_M f d(P_t^*\nu) = \int_M P_tf d\nu.
    \end{equation*}
    Then we have
    \begin{equation*}
        \frac{\partial}{\partial t}P_t^* = L^*P_t^* = P_t^*L^*,
    \end{equation*}
    called the forward Kolmogorov equation/Fokker-Planck equation, where $L^*$ acts on $\nu$ by
    \begin{equation*}
        \int_M f d(L^*\nu) = \int_M Lf d\nu.
    \end{equation*}
\end{rmk}

\section{Deterministic Flow} 

Let $V(t,x)$ be a time-dependent $C^\infty$-vector field. Consider ODE on $M$
\begin{equation}\label{eq:flowODE}
    \left\{
        \begin{aligned}
            &\frac{dx_t}{dt} = V(t,x_t) \\
            &x_0 = x
        \end{aligned}
    \right.
\end{equation}
and denote $\varphi_t(x)$ be its flow.

\begin{thm}
    Let $\bb{\varphi_t}_{t \in [0,T)}$ be a locally Lipschitz family of diffeomorphisms of $M$ with $\varphi_0 = id$. Let $V(t,x)$ be the time-dependent vector fields associated with $\bb{\varphi_t}_{t \in [0,T)}$. Let $\nu$ be a probability measure on $M$ and $\nu_t = (\varphi_t)_{\#}\nu$. Then $\nu_t$ is the unique solution of the following PDE 
    \begin{equation}\label{eq:conservation_mass}
        \left\{
            \begin{aligned}
                & \frac{\partial \nu_t}{\partial t} + \op{div}(\nu_tV_t) = 0,\quad t \in [0,T)\\
                & \nu_0 = \nu
            \end{aligned}
        \right.
    \end{equation}
    in $C([0,T),\mathcal{P}(M))$, where $\mathcal{P}(M)$ is equipped with the weak topology.
\end{thm}
\begin{rmk}
    For the measure PDE, it means that for any $f \in C_c^\infty(M)$,
    \begin{equation*}
        \inn{\partial_t \nu_t,f}+\inn{\op{div}(\nu_tV_t),f} = 0,
    \end{equation*}
    where
    \begin{align*}
        \inn{\partial_t \nu_t,f} &= \frac{\partial}{\partial t}\inn{\nu_t,f} = \frac{\partial}{\partial t} \int_M fd\nu_t, \\
        \inn{\op{div}(\nu_tV_t),f} &= -\int_M f d(\op{div}(\nu_tV_t)) = -\int_M \op{div}(f V_t) d\nu_t = \int_M \inn{\nabla f,V_t} d\nu_t.
    \end{align*}
    Moreover, if $d\nu_0 = pdm$ for $p \in C^\infty$, which implies there are $p_t \in C^\infty$ such that $d\nu_t = p_t dm$, then the PDE is equivalent to
    \begin{equation*}
        \left\{
            \begin{aligned}
                & \frac{\partial p_t}{\partial t} + \op{div}(p_tV_t) = 0,\quad t \in [0,T)\\
                & p_0 = p.
            \end{aligned}
        \right.
    \end{equation*}
\end{rmk}
The theorem implies that if there is a family of probability measures $\bb{\nu_t}_{t \in [0,T)}$ satisfies above PDE (\ref{eq:conservation_mass}), then for a random process $\widetilde{X}_t$ on $M$ satisfying the ODE (\ref{eq:flowODE}) with $\widetilde{X}_0 \sim \nu$, we have $\widetilde{X}_t \sim \nu_t$.

Consider the Riemannian Langevin dynamics (\ref{eq:RLD}) with $X_0 \sim \nu$, where $d\nu = p dm$ for a smooth $p$. Then $X_t \sim \nu_t$, for $d\nu_t = d(P_t^*\nu) = p_t dm$. By the Fokker-Planck equation, we have
\begin{equation*}
    \frac{\partial \nu_t}{\partial t} = L^*\nu_t,
\end{equation*}
which is equivalent to
\begin{equation*}
    \frac{\partial p_t}{\partial t} = \Delta_Mp_t + \op{div}(p_t\nabla U) = \op{div}(p_t(\nabla U + \nabla \log p_t)).
\end{equation*}
Therefore, if $\widetilde{X}_t$ satisfies
\begin{equation*}
    \left\{
        \begin{aligned}
            &\frac{d \widetilde{X}_t}{dt} = -\nabla U(\widetilde{X}_t) -\nabla \log p_t(\widetilde{X}_t), \\
            &\widetilde{X}_0 \sim \nu,
        \end{aligned}
    \right.
\end{equation*}
then $\widetilde{X}_t \sim \nu_t$, i.e., $\widetilde{X}_t \stackrel{\mathrm{d}}{=} X_t$.

\section{Convergence to Invariant Measure}

\paragraph{KL Divergence.} If we have
\begin{equation*}
    \op{Ric}_U = \op{Ric} + \nabla^2 U \geq \rho g,
\end{equation*}
for some $\rho > 0$, then above process satisfies the $\op{CD}(\rho,\infty)$ and so $\op{LSI}(1/\rho)$, which implies that
\begin{equation*}
    \op{End}_\mu(P_tg) \leq e^{-2\rho t}\op{Ent}_\mu(g),\quad \forall~g\in L^1(\mu).
\end{equation*}
(see note \emph{Stochastic Analysis}). If $X_0 \sim \nu_0$ with $d\nu_0 = f d\mu$ (assume $f \in L^1(\mu)$) and $\op{KL}(\nu_0 ~\Vert~ \mu) < \infty$, then
\begin{equation*}
    X_t \sim \nu_t = P_t^*\nu_0,
\end{equation*}
where $d(P_t^*\nu_0) = P_tf d\mu$. Then we can get
\begin{equation*}
    \op{KL}(\nu_t ~\Vert~ \mu) \leq e^{-2\rho t} \op{KL}(\nu_0 ~\Vert~ \mu).
\end{equation*}
Therefore,
\begin{equation*}
    \op{KL}(\nu_t ~\Vert~ \mu) \leq \mathcal{O}(e^{-t}).
\end{equation*}
On the other hand, by Pinsker inequality $d_{\op{TV}}(\nu_t,\mu) \leq \frac{1}{2}\op{KL}(\nu_t ~\Vert~ \mu)$,
\begin{equation*}
    d_{\op{TV}}(\nu_t,\mu) \leq \mathcal{O}(e^{-t}).
\end{equation*}

\paragraph{Wasserstein Distance.} Under the same condition,
\begin{equation*}
    \op{Ric}_U = \op{Ric} + \nabla^2 U \geq \rho g,
\end{equation*}
we can have
\begin{equation*}
    W_2(\nu_t,\mu) \leq e^{-\rho t}W_2(\nu_0,\mu).
\end{equation*}
\begin{rmk}
    It needs techniques from the optimal transport theory, like $W_2$-geodesic convexity, et. al., or it can be obtained by directly calculating from the SDE.
\end{rmk}
