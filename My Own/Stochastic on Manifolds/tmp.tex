\chapter{Riemannian Langevin Dynamics}

Let $(M,g)$ be a complete Riemannian manifold with Levi-Civita connection $\nabla$ and the canonical Riemannian volume measure $dm$. In the following, we denote $\op{grad} f = \nabla f$ for convenience. Let $B^M$ is a Brownian motion on $M$. Let $U \in C^\infty(M)$. Consider the following SDE
\begin{equation}\label{eq:RLD}
    dX_t = -\nabla U(X_t) dt + \sqrt{2}dB^M_t,
\end{equation}
which is called a Riemannian Langevin dynamics.

\section{Diffusion Operator}

Because $B^M$ is a Brownian motion on $M$, its anti-development $W$ is a standard Euclidean Brownian motion. So
\begin{equation*}
    dB^M_t = V_\alpha \circ dW^\alpha_t,
\end{equation*}
with
\begin{equation*}
    \Delta_M = \sum_\alpha V_\alpha^2.
\end{equation*}
Therefore,
\begin{equation*}
    dX_t = -\nabla U(X_t) dt + \sqrt{2}V_\alpha \circ dW^\alpha_t,
\end{equation*}
which implies that the diffusion operator of $X$ is
\begin{align*}
    Lf &= \frac{1}{2} \sum_\alpha (\sqrt{2}V_\alpha)^2 f - \nabla U (f) \\
    &= \Delta_M f - \inn{\nabla U,\nabla f},
\end{align*}
i.e., $L = \Delta_M - \inn{\nabla U, \nabla \cdot}$. For such $L$, in note \emph{Stochastic Analysis}, we have shown that 
\begin{equation*}
    L^*p = \Delta_M p + \op{div}(p\nabla U)
\end{equation*}
in the sense of on $L^2(dm)$. Also,
\begin{equation*}
    d\mu = e^{-U}dm,
\end{equation*}
is the invariant measure of such diffusion process, which is also symmetric for $L$. Note that $L$ is essentially self-adjoint on $C^\infty_c(M)$ w.s.t. $\mu$. Therefore, it has a unique self-adjoint extension $L$ on $L^2(\mu,M)$. Let
\begin{equation*}
    P_t = e^{-tL}
\end{equation*}
be the functional calculus of $L$, defined on $L^2(\mu,M)$. On the other hand, define
\begin{equation*}
    Q_t f(x) = \E\bj{f(X_t) \mid X_0 = x},\quad \forall~ f \in C_c^\infty(M),
\end{equation*} 
By note \emph{Markov Operator}, $Q_t$ can be extended to $L^2(\mu)$ and has the generator $A$ that is self-adjoint on $L^2(\mu)$. By Dynkin's formula,
\begin{equation*}
    Af = Lf,\quad \forall~f \in C_c^\infty(M).
\end{equation*}
Therefore, $L \subset A$. By the essential self-adjointness of $L$, $L = A$, which implies that
\begin{equation*}
    P_tf(x) = Q_tf(x) = \E\bj{f(X_t) \mid X_0 = x},\quad \forall~f \in L^2(\mu).
\end{equation*}
\begin{rmk}
    Note that if $\nabla U$ is globally Lipschitz continuous (in particular, $M$ is compact), i.e., $\abs{\nabla^2 U} \leq L$, then it can show that $(Q_t)_{t \geq 0}$ is a Feller-Dynkin semigroup so that it can be defined on $C_0(M)$.
\end{rmk}
\begin{rmk}
    For $P_t$, note that
    \begin{equation*}
        \frac{\partial}{\partial t}P_t = LP_t = P_tL,
    \end{equation*}
    called the backward Kolmogorov equation. Moreover, it can define the Markov transition kernel
    \begin{equation*}
        p_t(x,A) = P_t\mathbb{I}_A(x) = \E\bj{\mathbb{I}_A(X_t) \mid X_0 = x} = \Pb_x(X_t \in A).
    \end{equation*}
    Then
    \begin{equation*}
        P_tf(x) = \int_M f(y)p_t(x,dy).
    \end{equation*}
    It follows that we can define
    \begin{equation*}
        \int_M f d(P_t^*\nu) = \int_M P_tf d\nu.
    \end{equation*}
    Then we have
    \begin{equation*}
        \frac{\partial}{\partial t}P_t^* = L^*P_t^* = P_t^*L^*,
    \end{equation*}
    called the forward Kolmogorov equation/Fokker-Planck equation, where $L^*$ acts on $\nu$ by
    \begin{equation*}
        \int_M f d(L^*\nu) = \int_M Lf d\nu,\quad \forall~f \in \mathcal{D}(L).
    \end{equation*}
    Note that $L^*\nu$ may not be a measure. Note that $\mu$ is an invariant measure if and only if $L^*\mu = 0$, i.e.,
    \begin{equation*}
        \int_M Lf d\mu = 0,\quad \forall~f \in C_c^\infty(M).
    \end{equation*}
\end{rmk}

\section{Deterministic Flow} 

Let $V(t,x)$ be a time-dependent $C^\infty$-vector field. Consider ODE on $M$
\begin{equation}\label{eq:flowODE}
    \left\{
        \begin{aligned}
            &\frac{dx_t}{dt} = V(t,x_t) \\
            &x_0 = x
        \end{aligned}
    \right.
\end{equation}
and denote $\varphi_t(x)$ be its flow.

\begin{thm}
    Let $\bb{\varphi_t}_{t \in [0,T)}$ be a locally Lipschitz family of diffeomorphisms of $M$ with $\varphi_0 = id$. Let $V(t,x)$ be the time-dependent vector fields associated with $\bb{\varphi_t}_{t \in [0,T)}$. Let $\nu$ be a probability measure on $M$ and $\nu_t = (\varphi_t)_{\#}\nu$. Then $\nu_t$ is the unique solution of the following PDE 
    \begin{equation}\label{eq:conservation_mass}
        \left\{
            \begin{aligned}
                & \frac{\partial \nu_t}{\partial t} + \op{div}(\nu_tV_t) = 0,\quad t \in [0,T)\\
                & \nu_0 = \nu
            \end{aligned}
        \right.
    \end{equation}
    in $C([0,T),\mathcal{P}(M))$, where $\mathcal{P}(M)$ is equipped with the weak topology.
\end{thm}
\begin{rmk}
    For the measure PDE, it means that for any $f \in C_c^\infty(M)$,
    \begin{equation*}
        \inn{\partial_t \nu_t,f}+\inn{\op{div}(\nu_tV_t),f} = 0,
    \end{equation*}
    where
    \begin{align*}
        \inn{\partial_t \nu_t,f} &= \frac{\partial}{\partial t}\inn{\nu_t,f} = \frac{\partial}{\partial t} \int_M fd\nu_t, \\
        \inn{\op{div}(\nu_tV_t),f} &= -\int_M f d(\op{div}(\nu_tV_t)) = -\int_M \op{div}(f V_t) d\nu_t = \int_M \inn{\nabla f,V_t} d\nu_t.
    \end{align*}
    Moreover, if $d\nu_0 = pdm$ for $p \in C^\infty$, which implies there are $p_t \in C^\infty$ such that $d\nu_t = p_t dm$, then the PDE is equivalent to
    \begin{equation*}
        \left\{
            \begin{aligned}
                & \frac{\partial p_t}{\partial t} + \op{div}(p_tV_t) = 0,\quad t \in [0,T)\\
                & p_0 = p.
            \end{aligned}
        \right.
    \end{equation*}
\end{rmk}
The theorem implies that if there is a family of probability measures $\bb{\nu_t}_{t \in [0,T)}$ satisfies above PDE (\ref{eq:conservation_mass}), then for a random process $\widetilde{X}_t$ on $M$ satisfying the ODE (\ref{eq:flowODE}) with $\widetilde{X}_0 \sim \nu$, we have $\widetilde{X}_t \sim \nu_t$.

Consider the Riemannian Langevin dynamics (\ref{eq:RLD}) with $X_0 \sim \nu$, where $d\nu = p dm$ for a smooth $p$. Then $X_t \sim \nu_t$, for $d\nu_t = d(P_t^*\nu) = p_t dm$. By the Fokker-Planck equation, we have
\begin{equation*}
    \frac{\partial \nu_t}{\partial t} = L^*\nu_t,
\end{equation*}
which is equivalent to
\begin{equation*}
    \frac{\partial p_t}{\partial t} = \Delta_Mp_t + \op{div}(p_t\nabla U) = \op{div}(p_t(\nabla U + \nabla \log p_t)).
\end{equation*}
Therefore, if $\widetilde{X}_t$ satisfies
\begin{equation*}
    \left\{
        \begin{aligned}
            &\frac{d \widetilde{X}_t}{dt} = -\nabla U(\widetilde{X}_t) -\nabla \log p_t(\widetilde{X}_t), \\
            &\widetilde{X}_0 \sim \nu,
        \end{aligned}
    \right.
\end{equation*}
then $\widetilde{X}_t \sim \nu_t$, i.e., $\widetilde{X}_t \stackrel{\mathrm{d}}{=} X_t$.

\section{Convergence to Invariant Measure}

\paragraph{KL Divergence.} If we have
\begin{equation*}
    \op{Ric}_U = \op{Ric} + \nabla^2 U \geq \rho g,
\end{equation*}
for some $\rho > 0$, then above process satisfies the $\op{CD}(\rho,\infty)$ and so $\op{LSI}(1/\rho)$, which implies that
\begin{equation*}
    \op{End}_\mu(P_tg) \leq e^{-2\rho t}\op{Ent}_\mu(g),\quad \forall~g\in L^1(\mu).
\end{equation*}
(see note \emph{Stochastic Analysis}). If $X_0 \sim \nu_0$ with $d\nu_0 = f d\mu$ (assume $f \in L^1(\mu)$) and $\op{KL}(\nu_0 ~\Vert~ \mu) < \infty$, then
\begin{equation*}
    X_t \sim \nu_t = P_t^*\nu_0,
\end{equation*}
where $d(P_t^*\nu_0) = P_tf d\mu$. Then we can get
\begin{equation*}
    \op{KL}(\nu_t ~\Vert~ \mu) \leq e^{-2\rho t} \op{KL}(\nu_0 ~\Vert~ \mu).
\end{equation*}
Therefore,
\begin{equation*}
    \op{KL}(\nu_t ~\Vert~ \mu) \leq \mathcal{O}(e^{-t}).
\end{equation*}
On the other hand, by Pinsker inequality $d_{\op{TV}}(\nu_t,\mu) \leq \frac{1}{2}\op{KL}(\nu_t ~\Vert~ \mu)$,
\begin{equation*}
    d_{\op{TV}}(\nu_t,\mu) \leq \mathcal{O}(e^{-t}).
\end{equation*}

\paragraph{Wasserstein Distance.} Under the same condition,
\begin{equation*}
    \op{Ric}_U = \op{Ric} + \nabla^2 U \geq \rho g,
\end{equation*}
we can have
\begin{equation*}
    W_2(\nu_t,\mu) \leq e^{-\rho t}W_2(\nu_0,\mu).
\end{equation*}
\begin{rmk}
    It needs techniques from the optimal transport theory, like $W_2$-geodesic convexity, et. al., or it can be obtained by directly calculating from the SDE.
\end{rmk}

\section{Time-reversal Process}

This section is based on results in \cite{de2022riemannian}. Let $(M,g)$ be a compact Riemannian manifold (without boundary) with Levi-Civita connection $\nabla$ and canonical volume measure $dm$ and the corresponding probability measure $dp_{\op{ref}} = dm / m(M)$. In be following, we consider a more general equation.
\begin{equation*}
    dX_t = b(X_t)dt + dB^M_t,\quad t \in [0,T]
\end{equation*}
where $b \in \Gamma(TM)$, $B^M$ is a Brownian motion on $M$. The main goal of this section is to consider its time-reversal process $Y_t = X_{T-t}$. It can show that \begin{equation*}
    dY_t = (-b(Y_t) + \nabla \log p_{T-t}(Y_t))dt + dB^M_t,
\end{equation*}
where $X_t \sim \mu_t$ with $d\mu_t = p_t dp_{\op{ref}}$. It is equivalent to prove that for any $f \in C^\infty(M)$,
\begin{equation*}
    M^{Y,f}_t \defeq f(Y_t) - f(Y_0) - \int_0^t \inn{-b(Y_s) + \nabla \log p_{T-s}(Y_s),\nabla f(Y_s)} + \frac{1}{2}\Delta_Mf(Y_s)ds
\end{equation*}
is a martingale, by the weak uniqueness of Martingale problem. Furthermore, it suffices to show that for any $g \in C^\infty(M)$ and any $t \geq s \in [0,T]$,
\begin{equation*}
    \E\bj{g(Y_s)(f(Y_t)-f(Y_s))} = \E\bj{g(Y_s)\int_s^t\tilde{\mathcal{A}}_sf(Y_s)ds},
\end{equation*}
where
\begin{equation*}
    \tilde{\mathcal{A}}_tf = \inn{-b + \nabla \log p_{T-t}, \nabla f} + \frac{1}{2}\Delta_M f.
\end{equation*}
Note that for $t \geq s \in [0,T]$
\begin{align*}
    \E\bj{g(Y_s)(f(Y_t)-f(Y_s))} &= \E\bj{g(X_{T-s})(f(X_{T-t})-f(X_{T-s}))} \\
    &= \E\bj{\E[g(X_{T-s}) \mid X_{T-t}]f(X_{T-t})} - \E[g(X_{T-s})f(X_{T-s})] \\
    &= \E\bj{v(T-t,X_{T-t})f(X_{T-t})}- \E\bj{v(T-s,X_{T-s})f(X_{T-s})},
\end{align*}
where $v \colon [0,T-s] \times M \sto \R$ is given by
\begin{equation*}
    v(u,x) = \E[g(X_{T-s}) \mid X_u = x] = P_{T-s-u}g(x),
\end{equation*}
where $P_t$ is the semigroup associated with $X_t$. Assume $v$ is regular enough. So it satisfies the backward Kolmogorov equation
\begin{equation*}
    \partial_u v(u,x) = -\mathcal{A}v(u,x).
\end{equation*}
Let $h \colon [0,T-s] \times M \sto \R$ be defined as
\begin{equation*}
    h(u,x) = v(u,x)f(x).
\end{equation*}
Note that by It\^o's formula, because $\mathcal{A}$ is the generator for $X$,
\begin{equation*}
     M^{X,h}_t = h(t,X_t) - h(0,X_0) - \int_0^t (\partial_u + \mathcal{A})h(u,X_u)du
\end{equation*}
is martingale. It implies that
\begin{equation*}
    \E\bj{v(T-t,X_{T-t})f(X_{T-t})}- \E\bj{v(T-s,X_{T-s})f(X_{T-s})} = \int_{T-s}^{T-t}\E\bj{(\partial_u + \mathcal{A})h(u,X_u)}du.
\end{equation*}
To calculate it, first, we have
\begin{align*}
    \partial_u h(u, x)+\mathcal{A} h(u, x)&=f(x) \partial_u v(u, x)+f(x) \mathcal{A} v(u, x)+v(u, x) \mathcal{A} f(x)+\langle\nabla f(x), \nabla v(u, x)\rangle \\
    &=v(u, x) \mathcal{A} f(x)+\langle\nabla f(x), \nabla v(u, x)\rangle,
\end{align*}
because $\Delta_M(fg) = f \Delta_Mg+g\Delta_Mf + 2\inn{\nabla f, \nabla g}$ by $\op{div}(fX) = \inn{\nabla f, X}+f\op{div}X$. By the divergence theorem,
\begin{equation*}
    \E\bj{\langle\nabla f(X_u), \nabla v(u, X_u)\rangle} = -\mathbb{E}\left[v\left(u, X_u\right) \Delta_M f\left(X_u\right)\right]-\mathbb{E}\left[v\left(u, X_u\right)\left\langle\nabla f\left(X_u\right), \nabla \log p_u\left(X_u\right)\right\rangle\right] .
\end{equation*}
Therefore, it implies that
\begin{align*}
    \mathbb{E}\left[(\partial_u +\mathcal{A})h\left(u, X_u\right)\right] &= \mathbb{E}\left[v\left(u, X_u\right)\left(\left\langle b\left(X_u\right)-\nabla \log p_u\left(X_u\right), \nabla f\left(X_u\right)\right\rangle-\frac{1}{2} \Delta_{M} f\left(X_u\right)\right)\right] \\
    &= -\mathbb{E}\left[v\left(u, X_u\right) \tilde{\mathcal{A}}_{T-u}f\left(X_u\right)\right].
\end{align*}
Therefore, we have
\begin{align*}
    \E\bj{g(Y_s)(f(Y_t)-f(Y_s))} &= \int_{T-t}^{T-s} \mathbb{E}\left[v\left(u, X_u\right) \tilde{\mathcal{A}}_{T-u}f\left(X_u\right)\right] du \\
    &=\int_{T-t}^{T-s} \mathbb{E}\left[\E\bj{g(X_{T-s}) \mid X_u} \tilde{\mathcal{A}}_{T-u}f\left(X_u\right)\right] du \\
    &=\int_{T-t}^{T-s} \mathbb{E}\left[\E\bj{g(X_{T-s})\tilde{\mathcal{A}}_{T-u}f\left(X_u\right) \mid X_u} \right] du \\
    &= \int_{T-t}^{T-s} \E\bj{g(X_{T-s})\tilde{\mathcal{A}}_{T-u}f\left(X_u\right) } du \\
    &= \E\bj{g(Y_s)\int_{s}^{t}\tilde{\mathcal{A}}_{u}f\left(Y_u\right) du}.
\end{align*}

\paragraph{Integration by Parts.} Let $(\mathsf{X},\mathcal{X})$ be a polished space. Let $\Pb$ be a probability measure on $\Omega = C([0,T],\mathsf{X})$ equipping with canonical measurable sets. Let $X$ be the canonical process valued on $\mathcal{X}$ defined on $\Omega$, i.e., $X \colon C([0,T],\mathsf{X}) \sto \mathsf{X}$,
\begin{equation*}
    X_t(\omega) \defeq \omega_t,
\end{equation*}
and $\mathbb{F}^X = (\mathcal{F}_t)_{t \in [0,T]}$ with $\mathcal{F}_t = \sigma(X_s \colon s \in [0,t])$. For any $t \in [0,T]$, $\Pb_t \defeq (X_t)_{\#}\Pb$ and more general, $\Pb_{\mathcal{T}} = (X_{\mathcal{T}})_{\#}\Pb$.

Let $M = (M_t)_{t \in [0,T]}$ be a $\mathsf{X}$-valued and $\mathbb{F}^X$-adapted process on $\Omega$, which is called a $\Pb$-local martingale if it is a local martingale w.s.t. $\mathbb{F}^X$. 

For such $\Pb$, it can define the extended generator $\bar{\mathcal{A}}_\Pb$ (No Markov in advance). For a function,
\begin{equation*}
    u \colon [0,T] \times \mathsf{X} \sto \R
\end{equation*}
if there exists a $\mathsf{X}$-valued process $A = (A_t)_{t \in[0,T]}$ on $\Omega$ such that
\begin{enumerate}[label=(\alph{*})]
    \item $A$ is $\mathbb{F}^X$-adapted,
    \item $\int_0^T \abs{A_s} ds < \infty$ $\Pb$-a.e.,
    \item $M = (M_t)_{t \in [0,T]}$ defined as
    \begin{equation*}
        M_t = u(t,X_t) - u(0,X_0) - \int_0^tA_sds,
    \end{equation*}
    is a $\Pb$-local martingale.
\end{enumerate}
Note that if $A$ exists, then it is unique, because
\begin{equation*}
    \int_0^t A_s - B_s ds \text{ a local martingale }\quad \Rightarrow \quad A_t = B_t
\end{equation*}
$dt \otimes \Pb$-a.e.. Moreover, because $A$ is $\left(\sigma\left(X_{[0, t]}\right) ; 0 \leq t \leq T\right)$-adapted, i.e., $A_t$ is $\sigma(Y)$-measurable, for $Y = X_{[0,t]}$, $A_t = A_t(Y) = A_t(X_{[0,t]})$. Then we say $u \in \mathcal{D}(\bar{\mathcal{A}}_\Pb)$ and 
\begin{equation*}
    A_t \eqdef \bar{\mathcal{A}}_\Pb u(t, X_{[0,t]}).
\end{equation*}
Let $(u,v)$ be $u,v \colon [0,T] \times \mathsf{X} \sto \R$ such that $u,v,uv \in \mathcal{D}(\bar{\mathcal{A}}_\Pb)$. For $(u,v)$, define the carr\'e du champ $\bar{\Upsilon}_{\mathbb{P}}$
\begin{equation*}
    \bar{\Upsilon}_{\mathbb{P}} = \bar{\mathcal{A}}_\Pb(uv) - u\bar{\mathcal{A}}_\Pb(v) - v\bar{\mathcal{A}}_\Pb(u).
\end{equation*}
Assume there exists an algebra $\mathcal{U}_\Pb \subset \mathcal{D}(\bar{\mathcal{A}}_\Pb) \cap C_b(X)$. Let
\begin{equation*}
    \mathcal{U}_{\mathbb{P}, 2}=\left\{u \in \mathcal{U}_{\mathbb{P}}: \bar{\mathcal{A}}_\Pb u \in \mathrm{~L}^2(\mathbb{P}), \bar{\Upsilon}_{\mathbb{P}}(u, u) \in \mathrm{L}^1(\mathbb{P})\right\} .
\end{equation*}

If $\Pb$ is Markov, it has the usual operator $\mathcal{A}_\Pb u \colon [0,T] \times \mathsf{X} \sto \R$. We can see
\begin{equation*}
    \bar{\mathcal{A}}_\Pb u(t, X_{[0,t]}) =  \mathcal{A}_\Pb u(t,X_t)
\end{equation*}
For $\Pb$, let $R\Pb$ be the distribution of the inverse process of $X$, i.e., $t \mapsto X_{T - t}$. If $\Pb$ is Markov, then $R\Pb$ is also a Markov. In the following, we only consider the Markov case. For $u \colon \mathsf{X} \sto \R$ and $u \in \mathcal{D}(\mathcal{A}_{\Pb})$, we denote
\begin{equation*}
    \mathcal{A}_{\Pb}u(t,x) \eqdef \mathcal{A}_{\Pb,t}u(x).
\end{equation*}
For Markov case, we use $\mathcal{U}_\Pb$, $\mathcal{U}_{\mathbb{P}, 2}$ denotes time-independent $u$ such that the conditions are satisfied for all $t \in [0,T]$.

\begin{thm}[Theorem 3.17,~\cite{cattiaux2023time}]
    Let $u,v \in \mathcal{U}_{\mathbb{P}, 2}$.
    \begin{enumerate}[label=(\alph{*})]
        \item If the following holds,
        \begin{enumerate}[label=(\roman*)]
            \item $(t,x) \mapsto \Upsilon_{\Pb,t}(u,v)(x)$ is continuous.
            \item $\mathcal{U}_{\mathbb{P}, 2}$ determines the weak convergence of Borel measures on $\mathcal{X}$.
            \item the linear map
            \begin{equation*}
                v \in \clo{\mathcal{U}}_{2,\Pb} \sto \E\bj{\int_0^T \Upsilon_{\Pb,t}(u,v_t)(X_t)dt}
            \end{equation*}
            determines a finite measure on $[0,T] \times \mathsf{X}$, where
            \begin{equation*}
                \clo{\mathcal{U}}_{2,\Pb} = \bb{v \in C([0,T] \times \mathsf{X},\R) \colon v(t,\cdot) \in \mathcal{U}_{\mathbb{P}, 2},~\forall~t \in [0,T]}.
            \end{equation*}
        \end{enumerate}
        then $u \in \mathcal{D}(\mathcal{A}_{R(\Pb)})$ and $\mathcal{A}_{R(\Pb)} u \in \mathrm{L}^1(\Pb)$. 

        \item If $u \in \mathcal{D}(\mathcal{A}_{R(\Pb)})$ and $\mathcal{A}_{R(\Pb)} u \in \mathrm{L}^1(\Pb)$, then for a.e. $t \in [0,T]$,
        \begin{equation*}
            \E\bj{\bc{\mathcal{A}_{\Pb,t} u(X_t) + \mathcal{A}_{R(\Pb),T-t}u(X_t)}v(X_t) + \Upsilon_{\Pb,t}(u,u)(X_t)} = 0.
        \end{equation*}
    \end{enumerate}
\end{thm}
 
\paragraph{Girsanov Theory on Compact Riemannian Manifolds.} Let $(M,g)$ be a compact Riemannian manifold and isometrically and properly embedding in $\R^p$.

\begin{thm}
    Let $\mathbb{Q}$ be the path measure of a Brownian motion on $M$. Let $\Pb$ be a Markov path measure on $C([0,T],M)$ such that $\op{KL}(\Pb ~\Vert~ \mathbb{Q} ) < \infty$. Then there exists a time-dependent vector field $\beta$ such that $\Pb$ satisfies the martingale problem for generator $\mathcal{A}$, where for $t \in [0,T]$, $u \in C^2(M)$, and $x \in M$,
    \begin{equation*}
        \mathcal{A}(t,u)(x) = \inn{\beta(t,x),\nabla u(x)} + \frac{1}{2}\Delta_Mu(x),
    \end{equation*}
    In addition, we have 
    \begin{equation*}
        \op{KL}(\Pb ~\Vert~ \mathbb{Q} ) = \op{KL}(\Pb_0 ~\Vert~ \mathbb{Q}_0 ) + \frac{1}{2}\int_0^T \E\bj{\norm{\beta(t,X_t)}^2}dt.
    \end{equation*}
\end{thm}
\begin{rmk}
    For Euclidean case, if $\mathbb{P} \ll \mathbb{Q}$, then for $X \sim \Pb$, we can find an Euclidean Brownian motion $W$ such that
    \begin{equation*}
        dX_t = v(t,X_t)dt + dW_t,
    \end{equation*}
    which implies that the generator
    \begin{equation*}
        \mathcal{A}_t = \inn{v,\nabla \cdot} + \frac{1}{2} \Delta \cdot.
    \end{equation*}
    Moreover, by Girsanov's theorem,
    \begin{equation*}
        \op{KL}(\Pb ~\Vert~ \mathbb{Q} ) = \op{KL}(\Pb_0 ~\Vert~ \mathbb{Q}_0 ) + \frac{1}{2}\int_0^T \E\bj{\norm{v(t,X_t)}^2}dt.
    \end{equation*}
\end{rmk}
\begin{proof}
    By embedding, if let $B_t^M$ be a Brownian motion on $M$, we have
    \begin{equation*}
        dB_t^M = P_i(B^M_t) \circ dB^i_t = P(B^M_t) \circ dB_t,
    \end{equation*}
    for the orthogonal projection $P$ and an Euclidean Brownian motion $B$. Note that
    \begin{equation*}
        dB_t^M = \bar{b}(B^M_t)dt +  P(B^M_t) dB_t,
    \end{equation*} 
    where $\bar{b}$ satisfies $P(x)\bar{b}(x) = 0$ for all $x \in M$. It implies that
    \begin{equation*}
        \frac{1}{2}\Delta_Mu(x) = \inn{\clo{\nabla} \bar{u}(x),\clo{\nabla} \bar{v}(x)} + \frac{1}{2}\inn{P(x),\clo{\nabla}^2 \bar{u}(x)},
    \end{equation*}
    for any $u,v \in C_c^2(M)$, where $\bar{u},\bar{v} \in C_c^2(\R^p)$ be their extensions. Because $\mathbb{Q}$ is the distribution of $B^M$, which can be viewed on $\R^p$, by uniqueness condition and finite entropy, there exists $\bar{\beta} \colon [0,T] \times \R^p \sto \R^p$ such that
    \begin{equation*}
        \op{KL}(\mathbb{P} ~\Vert~ \mathbb{Q})=\op{KL}\left(\mathbb{P}_0 ~\Vert~ \mathbb{Q}_0\right)+\frac{1}{2} \int_0^T \mathbb{E}\left[\left\|P\left(X_t\right) \bar{\beta}\left(t, X_t\right)\right\|^2\right] d t ,
    \end{equation*}
    and $\Pb$ satisfies the martingale problem for $\bar{A} \colon [0,T] \times C^2_c(\R^p) \sto \R$ with
    \begin{equation*}
        \bar{\mathcal{A}}(t, \bar{u})(x)=\langle\bar{b}(x)+ P(x) \bar{\beta}(t, x), \clo{\nabla} \bar{u}(x)\rangle+\frac{1}{2}\left\langle P(x), \clo{\nabla}^2 \bar{u}(x)\right\rangle .
    \end{equation*}
    Let $\beta = P\bar{\beta}$. Note that for any $u \in C_c^2(M)$, if $\bar{u}$ is a extension of $u$ constant-along-normals, then for any $X \in \Gamma(TM)$,
    \begin{equation*}
        \nabla^2 u(X,Y) = \clo{\nabla}^2\bar{u}(X,Y).
    \end{equation*}
    By $\Delta_M u = \sum_\alpha\nabla^2 u(P_\alpha,P_\alpha)$,
    \begin{align*}
        \sum_\alpha\nabla^2 u(P_\alpha,P_\alpha) &= \sum_\alpha\clo{\nabla}^2 \bar{u}(P_\alpha,P_\alpha) \\
        &= \sum_\alpha P_\alpha^iP_\alpha^j \frac{\partial^2 \bar{u}}{\partial x^i \partial x^j} \\
        &= \sum_{i,j}P_i^j \frac{\partial^2 \bar{u}}{\partial x^i \partial x^j} \\
        &= \inn{P,\clo{\nabla}^2 \bar{u}}
    \end{align*}
    because $P = P^2$. Moreover, $P(x)\bar{b}(x) = 0$ implies that $\bar{b} \in \Gamma(NM)$, which follows that $\inn{\bar{b}(x),\clo{\nabla} \bar{u}} = 0$. So
    \begin{align*}
        \bar{\mathcal{A}}(t, \bar{u})(x)&=\langle\bar{b}(x)+ P(x) \bar{\beta}(t, x), \clo{\nabla} \bar{u}(x)\rangle+\frac{1}{2}\left\langle P(x), \clo{\nabla}^2 \bar{u}(x)\right\rangle \\
        &=\langle\beta(t, x), \nabla u(x)\rangle+\frac{1}{2} \Delta_M u. \qedhere
    \end{align*}
    In particular, when viewing $\Pb$ as a process on $M$, it satisfies the martingale problem with generator $\mathcal{A} \colon [0,T] \times C_c^2(M) \times M \sto \R$ given by
    \begin{equation*}
        \mathcal{A}(t,u)(x) = \langle\beta(t, x), \nabla u(x)\rangle+\frac{1}{2} \Delta_M u(x). 
    \end{equation*}
\end{proof}


\begin{cor}
    Let $\Pb^1,\Pb^2$ be two Markov path measures on $C([0,T],M)$ with $\Pb^1_0 = \Pb^2_0$, associating with SDE
    \begin{equation*}
        dX^i_t = b_i(t,X^i_t)dt + dB^M_t.
    \end{equation*}
    Then we have
    \begin{equation*}
        \op{KL}(\Pb^1 ~\Vert~ \Pb^2) = \frac{1}{2}\int_0^T\E\bj{\norm{b_1(t,X^1_t) - b_2(t,X^2_t)}^2}dt.
    \end{equation*}
\end{cor}
\begin{proof}
    Let $M$ be a compact manifold isometrically embedded in $\R^p$ such that $M \subset B(0,R)$. Let $\varphi \in C^\infty(\R^p)$ such that $\varphi|_{\clo{B(0,R)}} = 1$ and $\varphi(x) = 0$ for $\abs{x} \geq R+1$. Let $\bar{b}_i \in C_c^2([0,T] \times \R^p,\R^b)$ be the extension of $b_i(t,x)$. Consider
    \begin{align*}
        d \bar{X}_t^i&= \varphi\left(\bar{X}_t^i\right)\bc{\bar{b}^i\left(t, \bar{X}_t^i\right)dt + dB^M_t} \\
        &=\varphi\left(\bar{X}_t^i\right)\left\{\mathrm{P}\left(\bar{X}_t^i\right) \bar{b}^i\left(t, \bar{X}_t^i\right)+\bar{b}\left(\bar{X}_t\right)\right\} d t+\varphi\left(\bar{X}_t^i\right) \mathrm{P}\left(\bar{X}_t^i\right) \mathrm{dB}_t.
    \end{align*}
    So for any $x \in M$, $f \in C^2_c(M)$ with extension $\bar{f} \in C_c^2(\R^p)$,
    \begin{align*}
        \bar{\mathcal{A}}_t^i(\bar{f})(x) & =\left\langle\bar{b}^i(t, x)+\bar{b}(x), \nabla \bar{f}(x)\right\rangle+\frac{1}{2}\left\langle\mathrm{P}(x), \nabla^2 \bar{f}(x)\right\rangle \\
        & =\left\langle b^i(t, x), \nabla f(x)\right\rangle+\frac{1}{2} \Delta_{M} f(x) .
    \end{align*}
    It follows that $\bar{X}^i$ viewing as a process on $M$ has the same generator as $X^i$. So $\bar{X}^i$ has the distribution $\Pb^i$. Moreover, denote $\Pb^i$ as $\bar{\Pb}^i$ when it is viewed as $\R^p$-process. Note that because the
    \begin{equation*}
        \varphi(x) = 0,~\forall~\norm{x} \geq R+1
    \end{equation*}
    the global Lipschitz condition of SDE $\bar{X}^i$ is satisfied. Moreover, by the compactness,
    \begin{align*}
        \left(d \bar{\mathbb{P}}^1 / d \bar{\mathbb{P}}^2\right) & \left(\left(\bar{{X}}_t^1\right)_{t \in[0, T]}\right)=\exp \left[\int_0^T\left\langle\bar{b}^1\left(t, \bar{{X}}_t^1\right)-\bar{b}^2\left(t, \bar{{X}}_t^1\right), \mathrm{P}\left(\bar{{X}}_t^1\right) d \bar{{X}}_t^1\right\rangle\right. \\
        & \left.-\frac{1}{2} \int_0^T\left\langle\bar{b}^1\left(t, \bar{{X}}_t^1\right)-\bar{b}^2\left(t, \bar{{X}}_t^1\right), \mathrm{P}\left(\bar{{X}}_t^1\right)\left(\bar{b}^1\left(t, \bar{{X}}_t^1\right)+\bar{b}^2\left(t, \bar{{X}}_t^1\right)\right)\right\rangle d t\right]\\
        &= \exp \left[\frac{1}{2} \int_0^T\left\|\bar{b}^1\left(t, \overline{{X}}_t^1\right)-\bar{b}^2\left(t, \overline{{X}}_t^1\right)\right\|^2 \mathrm{~d} t+\int_0^T\left\langle\bar{b}^1\left(t, \overline{{X}}_t^1\right)-\bar{b}^2\left(t, \overline{{X}}_t^1\right), \mathrm{P}\left(\overline{{X}}_t^1\right) \mathrm{d} {B}_t\right\rangle\right] .
    \end{align*}
    It follows that
    \begin{equation*}
        \op{KL}\left(\overline{\mathbb{P}}^1 ~\Vert~ \overline{\mathbb{P}}^2\right)=\frac{1}{2} \int_0^T \mathbb{E}\left[\left\|b^1\left(t, {X}_t^1\right)-b^2\left(t, {X}_t^1\right)\right\|^2\right] d t.\qedhere
    \end{equation*}
\end{proof}

\paragraph{Proof of Time-reversal Formula.}

\begin{cor}
    Let $\mathbb{Q}$ be the path measure of a Brownian motion on $M$ with $\mathbb{Q}_0 = p_{\op{ref}}$ and $\mathbb{P}$ be a path measure on $C([0,T],M)$ with $\op{KL}(\Pb ~\Vert~\mathbb{Q}) < \infty$. Then, there exist two time-dependent vector fields $\beta_\Pb$ and $\beta_{R(\Pb)}$ such that $\Pb$ and $R(\Pb)$ satisfy martingale problem associated with $\mathcal{A}_\Pb$ and $\mathcal{A}_{R(\Pb)}$,
    \begin{align*}
        & \mathcal{A}_{\mathbb{P}}(t, u)(x)=\left\langle\beta_{\mathbb{P}}(t, x), \nabla u(x)\right\rangle+\frac{1}{2} \Delta_{M} u(x), \\
        & \mathcal{A}_{R(\mathbb{P})}(t, u)(x)=\left\langle\beta_{R(\mathbb{P})}(t, x), \nabla u(x)\right\rangle+\frac{1}{2} \Delta_{M} u(x) .
    \end{align*}
    Finally, we have that
    \begin{equation*}
        \int_0^T \mathbb{E}\left[\left\|\beta_{\mathbb{P}}\left(t, {X}_t\right)\right\|^2\right] \mathrm{d} t+\int_0^T \mathbb{E}\left[\left\|\beta_{R(\mathbb{P})}\left(t, {X}_{T-t}\right)\right\|^2\right] \mathrm{d} t<+\infty,
    \end{equation*}
    for $X \sim \Pb$.
\end{cor}
\begin{proof}
    Just applying above theorem with
    \begin{equation*}
        \op{KL}(\Pb ~\Vert~ \mathbb{Q}) = \op{KL}(R(\Pb) ~\Vert~ R(\mathbb{Q})) = \op{KL}(R(\Pb) ~\Vert~ \mathbb{Q}) <\infty. \qedhere
    \end{equation*}
\end{proof}

Then by above the Integral by parts, for any $u,v \in C_c^\infty(M)$,
\begin{equation*}
    \mathbb{E}\left[v\left({X}_t\right)\left(\left\langle\beta_{\mathbb{P}}\left(t, {X}_t\right)+\beta_{R(\mathbb{P})}\left(T-t, {X}_t\right), \nabla u\left({X}_t\right)\right\rangle+\Delta_{M} u\left({X}_t\right)\right)+\left\langle\nabla u\left({X}_t\right), \nabla v\left({X}_t\right)\right\rangle\right]=0 .
\end{equation*}
It follows that
\begin{align*}
    &\int_{M}\left\{\left\langle\beta_{R(\mathbb{P})}(T-t, x), \nabla u(x)\right\rangle+\langle b(x), \nabla u(x)\rangle\right\} v(x) p_t(x) d p_{\text {ref }}(x) \\
    &\quad= \int_M \inn{\nabla \log p_t(x),\nabla u(x)}v(x)p_t(x)dp_{\text {ref }}(x).
\end{align*}
Therefore,
\begin{equation*}
    \left\langle\beta_{R(\mathbb{P})}(T-t, x), \nabla u(x)\right\rangle = \inn{\nabla \log p_t(x),\nabla u(x)}.
\end{equation*}


\section{Discretization}

Let $(M,g)$ be a complete Riemannian manifold and $B^M$ be a Brownian motion on it. Consider the SDE, 
$$ 
dX_t = v(X_t)dt + dB^M_t,~t \in [0,T],\quad X_0 = Z
$$ 
We always assume $B^M$ is independent of $Z$. Fix $h = T/N$ for some $N \in \N$. Let $t_k = kh$. Constructing process $X^h_k$ by 
$$
X^h_{k+1} = \exp_{X^h_k}(hv(X^h_k) + \sqrt{h}\xi), 
$$
where $\exp$ is the exponential map, and $\xi$ is a standard Gaussian standard Gaussian with respect to any orthonormal basis of $T_xM$. In the following, we study the convergence of $(X^h_k)_{k=0}^N$.

\paragraph{Weak convergence.} Assume $M$ has bounded geometry, that is,
        \begin{equation*}
            \op{inj}(M) \geq \iota > 0,\quad \sup_{x \in M} \abs{\nabla^k R(x)} < \infty,~\forall~k,
        \end{equation*}
        where $R$ is the Riemannian curvature. Let $(R_t)_{t \in [0,T]}$ be the semigroup of $X_t$ and $(R^h_k)_{k=0}^N$ be the semigroup of $X^h_k$. Assume that $v \in C^3_b(TM)$. Then for any $f \in C_b^3(M)$, we have
\begin{equation*}
    \norm{R_Tf - R_N^h f}_\infty \leq Ch.
\end{equation*}
\begin{proof}
    First, for $X_t$, the generator $L$ is
    \begin{equation*}
        Lf = \inn{v,\nabla f} + \frac{1}{2}\Delta_M f,\quad \forall~ f \in C_b^3(M),
    \end{equation*}
    and $R_T = (R_h)^N$. For $X^h_k$, let $R^h = R_1^h$. Then $R^h_N = (R^h)^N$.
    
    \vspace{0.5em}
    \textbf{Step 1.} Check: For any $f \in C^3_b(M)$, we have
    \begin{equation*}
        R^hf(x) = f(x) + hLf(x) + \mathcal{O}(h^{3/2}).
    \end{equation*}
    To prove that, first fix $X^h_0 = x$ with $x \in M$ and $f \in C_c^\infty(M)$. We can choose $0 < \rho \leq \iota$ (note that $\rho$ independent of $x$) such that
    \begin{equation*}
        \exp_x \colon B(0,\rho) \subset T_xM \rightarrow B(x,\rho) \subset M
    \end{equation*}
    is a diffeomorphism, moreover, we choose the normal coordinate on $T_xM$. Let
    \begin{equation*}
        Z_h(x) \defeq hv(x) + \sqrt{h}\xi,
    \end{equation*}
    so that
    \begin{equation*}
        X_1^h = \exp_x(Z_h(x)).
    \end{equation*}
    Define
    \begin{equation*}
        \varphi_x(z) \defeq f(\exp_x(z)),\quad \forall~ z \in B(0,\rho).
    \end{equation*}
    Because $f \in C_b^3(M)$ and $M$ has bounded geometry, 
    \begin{equation*}
        \sup _{x \in M} \sup _{|z| \leq \rho} \sum_{|\alpha| \leq 3}\left|\partial_z^\alpha \varphi_x(z)\right|<\infty
    \end{equation*}
    Note that
    \begin{equation*}
        R^hf(x) = \E\bj{f(X_1^h) \mid X_0^h = x} = \E\bj{\varphi_x(Z_h(x))}
    \end{equation*}
    \begin{enumerate}[label=\Roman{*}.]
        \item Fix $h_0 > 0$ sufficiently small such that
        \begin{equation*}
            h_0 \sup _x\|v(x)\| \leq \frac{\rho}{4}.
        \end{equation*}
        Then for $0 < h < h_0$, if $\abs{\xi} < {\rho}/{(2 \sqrt{h})}$, then
        \begin{equation*}
            \left|Z_h(x)\right| \leq h\|v(x)\|+\sqrt{h}|\xi| \leq \frac{\rho}{4}+\frac{\rho}{2}<\rho.
        \end{equation*}
        Therefore,
        \begin{equation*}
            A_h:=\left\{|\xi| \leq {\rho}/{(2 \sqrt{h})}\right\} \subset \bb{Z_h(x) \in B(0,\rho)}.
        \end{equation*}
        Moreover, because $\xi$ is Gaussian, then
        \begin{equation*}
            \Pb(A^c_h) = \Pb \bc{|\xi| > {\rho}/{(2 \sqrt{h})}} \leq c_1 e^{-c_2/h},
        \end{equation*}
        for some $c_1,c_2 > 0$. Writing
        \begin{equation*}
            R^h f(x)=\mathbb{E}\left[\varphi_x\left(Z_h(x)\right) \mathbb{I}_{A_h}\right]+\mathbb{E}\left[f\left(\exp _x\left(Z_h(x)\right)\right) \mathbb{I}_{A_h^c}\right] .
        \end{equation*}
        Then for the second term,
        \begin{equation*}
            \abs{\mathbb{E}\left[f\left(\exp _x\left(Z_h(x)\right)\right) \mathbb{I}_{A_h^c}\right]} \leq \norm{f}_\infty \Pb(A^c_h) \leq \norm{f}_\infty c_1 e^{-c_2/h} \leq \mathcal{O}(h^2).
        \end{equation*}
        So the main goal is to bound the first term.

        \item For $z \in B(0,\rho)$, by Taylor expansion
        \begin{equation*}
            \varphi_x(z)=  \varphi_x(0)+D \varphi_x(0)[z]+\frac{1}{2} D^2 \varphi_x(0)[z, z]  +R_3(x,z),
        \end{equation*}
        where
        \begin{equation*}
            R_3(x,z)=\frac{1}{6} \sum_{i, j, k} \partial_{i j k} \varphi_x(\theta z) z_i z_j z_k,
        \end{equation*}
        for some $\theta=\theta(x, z) \in(0,1)$. For $R_4$, it has
        \begin{equation*}
            \left|R_3(x, z)\right| \leq C_f|z|^3, \quad \forall~|z| \leq \rho .
        \end{equation*}
        for some $C_f > 0$ (independent of $x$).

        For $z = Z_h(x)$, writing
        \begin{align*}
            \mathbb{E}\left[\varphi_x\left(Z_h(x)\right) 1_{A_h}\right]= & \varphi_x(0) \mathbb{P}\left(A_h\right)+\mathbb{E}\left[D \varphi_x(0)\left[Z_h\right] 1_{A_h}\right] \\
            & +\frac{1}{2} \mathbb{E}\left[D^2 \varphi_x(0)\left[Z_h, Z_h\right] 1_{A_h}\right] \\
            & +\mathbb{E}\left[R_3(x,Z_h(x)) 1_{A_h}\right] 
        \end{align*}
        Next, consider these terms respectively.
        \begin{enumerate}[label=(\alph{*})]
            \item First, because
            \begin{equation*}
                \Pb(A_h) = 1 - \Pb(A_h^c) = 1 + \mathcal{O}\bc{e^{-c_1/h}},
            \end{equation*}
            the zero-order term can be bounded by
            \begin{equation*}
                \varphi_x(0) = \varphi_x(0) \mathbb{P}\left(A_h\right)=f(x)+\mathcal{O}\left(e^{-c_2 / h}\right) .
            \end{equation*}
            \item For the first-order term, we first recall that $Z_h(x)=h v(x)+\sqrt{h} \xi$ and $\E[\xi] = 0$. Because
            \begin{equation*}
               \norm{\mathbb{E}\left[Z_h(x) 1_{A^c_h}\right]} \leq \left(\mathbb{E}\left|Z_h(x)\right|^2\right)^{1 / 2}\left(\mathbb{P}\left(A_h^c\right)\right)^{1 / 2} \leq \mathcal{O}\left(e^{-c_3 / h}\right),
            \end{equation*}
            we have
            \begin{align*}
                \mathbb{E}\left[Z_h(x) 1_{A_h}\right] &= \mathbb{E}\left[Z_h(x)\right] - \mathbb{E}\left[Z_h(x) 1_{A^c_h}\right] \\
                &=\mathbb{E}\left[Z_h(x)\right]+\mathcal{O}\left(e^{-c_3 / h}\right) \\
                &=h v(x)+\mathcal{O}\left(e^{-c_3 / h}\right).
            \end{align*}
            Then we get
            \begin{equation*}
                \mathbb{E}\left[D \varphi_x(0)\left[Z_h\right] 1_{A_h}\right]=D \varphi_x(0)[h v(x)]+\mathcal{O}\left(e^{-c_2 / h}\right).
            \end{equation*}
            Because $\varphi_x(z) = f(\exp_x(z))$,
            \begin{equation*}
                D \varphi_x(0)[h v(x)] = h\inn{\nabla f(x), v(x)}.
            \end{equation*}
            So
            \begin{equation*}
                \mathbb{E}\left[D \varphi_x(0)\left[Z_h\right] 1_{A_h}\right]= h\inn{\nabla f(x), v(x)} + \mathcal{O}\left(e^{-c_2 / h}\right).
            \end{equation*}

            \item For the second-order term, first, we have
            \begin{equation*}
                D ^2\varphi_x(0)[z,z] = \nabla^2f(x)(z,z) = z^\top H_x z,
            \end{equation*}
            where $H_x = \nabla^2f(x)$ in matrix expression under the normal coordinate. Moreover, because $\xi \in \mathcal{N}(0,I)$,
            \begin{align*}
                \mathbb{E}\left[Z_h Z_h^{\top} 1_{A_h}\right] & =\mathbb{E}\left[Z_h Z_h^{\top}\right]+\mathcal{O}\left(e^{-c_2 / h}\right) \\
                & =\mathbb{E}\left[(h v+\sqrt{h} \xi)(h v+\sqrt{h} \xi)^{\top}\right]+\mathcal{O}\left(e^{-c_2 / h}\right) \\
                & =h^2 v v^{\top}+h \mathbb{E}\left[\xi \xi^{\top}\right]+\mathcal{O}\left(e^{-c_2 / h}\right) \\
                & =h \operatorname{Id}+h^2 v v^{\top}+\mathcal{O}\left(e^{-c_2 / h}\right) .
            \end{align*}
            So it implies that
            \begin{align*}
                \frac{1}{2} \mathbb{E}\left[D^2 \varphi_x(0)\left[Z_h, Z_h\right] 1_{A_h}\right] & =\frac{1}{2} \mathbb{E}\left[\operatorname{tr}\left(H_x Z_h Z_h^{\top}\right) 1_{A_h}\right] \\
                & =\frac{1}{2} \operatorname{tr}\left(H_x \mathbb{E}\left[Z_h Z_h^{\top} 1_{A_h}\right]\right) \\
                & =\frac{1}{2} h \operatorname{tr}\left(H_x\right)+\mathcal{O}\left(h^2\right)+\mathcal{O}\left(e^{-c_2 / h}\right) \\
                & =\frac{1}{2} h \Delta_M f(x)+\mathcal{O}\left(h^2\right) .
            \end{align*}

            \item For the remainder term, because $Z_h(x) = hv(x) + \sqrt{h}\xi$ with $\xi \sim \mathcal{N}(0,I)$,
            \begin{equation*}
                \E\bj{\norm{Z_h}^3} = \mathcal{O}(h^{3/2}).
            \end{equation*}
            Therefore,
            \begin{equation*}
                \left\|\mathbb{E}\left[R_3\left(x, Z_h(x)\right) 1_{A_h}\right]\right\| \leq C_f \mathbb{E}\bj{\left\|Z_h\right\|^3}=\mathcal{O}\left(h^{3/2}\right) .
            \end{equation*}
        \end{enumerate}

        Combining these results, we have
        \begin{equation*}
            R^hf(x) = f(x) + hLf(x) + \mathcal{O}(h^{3/2}),
        \end{equation*}
        which implies that
        \begin{equation*}
            \left\|R^h f-f-h L f\right\|_{\infty} \leq C_1 h^{3/2}.
        \end{equation*}
    \end{enumerate}

    \vspace{0.5em}
    \textbf{Step 2.} Check:
    \begin{equation*}
        \left\|R_h f-f-h L f\right\|_{\infty} \leq C_2 h^2.
    \end{equation*}
    Let $u(t,x) = R_tf(x)$. The by the backward Kolmogorov equation,
    \begin{equation*}
        \partial_t u_t = Lu_t,\quad u_0(x) = f(x).
    \end{equation*}
    By the smoothness and boundedness, one can show that $u_t \in C^3_b(M)$ and 
    \begin{equation}\label{eq:bound_parabolic}
        \sup _{0 \leq t \leq T}\left\|\nabla^\alpha u_t\right\|_{\infty} \leq C_3,\quad \forall~\abs{\alpha} \leq 3.
    \end{equation}
    In particular, $\norm{L^2 u_t}_\infty \leq C_3$.By
    \begin{equation*}
        u_h(x)-f(x)=\int_0^h \partial_t u_s(x) d s=\int_0^h L u_s(x) d s,
    \end{equation*}
    we get
    \begin{equation*}
        Lu_s(x)-Lf(x)=\int_0^s \partial_t Lu_r(x) d r=\int_0^s L^2 u_r(x) d r.
    \end{equation*}
    Therefore,
    \begin{align*}
        u_h(x) & =f(x)+h L f(x)+\int_0^h \int_0^s L^2 u_r(x) d r d s \\
        & =f(x)+h L f(x)+\beta_h(x)
    \end{align*}
    with
    \begin{equation*}
        \left|\beta_h(x)\right| \leq \int_0^h \int_0^s\left\|L^2 u_r\right\|_{\infty} d r d s \leq C_3 \int_0^h \int_0^s d r d s=\frac{1}{2} C_3 h^2.
    \end{equation*}
    It follows that
    \begin{equation*}
        \left\|R_h f-f-h L f\right\|_{\infty} \leq C_2 h^2.
    \end{equation*}

    \vspace{0.5em}
    \textbf{Step 3.} By Step 1 and Step 2, for all $f \in C_c^\infty$,
    \begin{align*}
        \left\|R^h f-R_h f\right\|_{\infty} & \leq\left\|R^h f-f-h L f\right\|_{\infty}+\left\|R_h f-f-h L f\right\|_{\infty} \\
        & \leq C_1 h^{3/2}+C_2 h^2 \leq C_4h^{3/2}.
    \end{align*}
    Globally, note that $R_T = (R_h)^N$ and $R^h_N = (R^h)^N$. Writing
    \begin{equation*}
        \left(R^h\right)^N-\left(R_h\right)^N=\sum_{k=0}^{N-1}\left(R^h\right)^{N-1-k}\left(R^h-R_h\right)\left(R_h\right)^k .
    \end{equation*}
    Because $R^h$ and $R_h$ are Markov, they are contractions. So
    \begin{align*}
        \left\|\left(R^h\right)^N f-\left(R_h\right)^N f\right\|_{\infty} \leq \sum_{k=0}^{N-1} \sup _x\left|\left(R^h-R_h\right)g_k(x)\right|,
    \end{align*}
    where by (\ref{eq:bound_parabolic})
    \begin{equation*}
        g_k(x) = \left(R_h^k f\right)(x) = \left(R_{kh} f\right)(x) \in C_b^3(M).
    \end{equation*}
    Therefore,
    \begin{align*}
        \left\|\left(R^h\right)^N f-\left(R_h\right)^N f\right\|_{\infty} &\leq \sum_{k=0}^{N-1} \left\|R^h g_k-R_h g_k\right\|_{\infty} \\
        &\leq \sum_{k=0}^{N-1} C_4h^{3/2} = C_4Nh^{3/2} = \frac{T}{h}C_4h^{3/2} = C_Th^{1/2}. \qedhere
    \end{align*}
\end{proof}
\begin{rmk}
    Above proof can be extended to time-inhomogeneous case
    \begin{equation*}
        dX_t = v(t,X_t)dt + g(t)dB^M_t.
    \end{equation*} 
\end{rmk}

\paragraph{Strong convergence.} Let $M$ be a complete Riemannian manifold and it is properly and isometrically embedded in $\R^n$. Assume that
\begin{enumerate}[label=(\Roman*)]
    \item\label{assum:conv_gem_1} the second fundamental form and its derivative are uniformly bounded, i.e.,
    \begin{equation*}
        \sup_{x \in M} \norm{\mathrm{II}_x}_{\op{op}} < \infty,~ \sup_{x \in M}\norm{\nabla\mathrm{II}_x}_{\op{op}} < \infty.
    \end{equation*}
    \item\label{assum:conv_gem_2} there exists a tubular neighborhood $V \subset \R^n$ of $M$, together with the canonical projection $\pi \colon V \sto M \subset \R^n$ such that
    \begin{equation*}
        \sup_{y \in V} \norm{D\pi(y)}_{\op{op}} < \infty.
    \end{equation*}
    \item\label{assum:conv_gem_3} $v \in \Gamma(TM)$ satisfies
    \begin{equation*}
        \sup_{x \in M} \norm{v(x)} < \infty,\quad \sup_{x \in M} \sup_{\norm{X} = 1} \norm{\nabla_X v(x)} < \infty.
    \end{equation*}
    \item\label{assum:conv_gem_4} $X_0 \in L^2$.
\end{enumerate}
Then we have the strong convergence of geometric EM algorithm, i.e.,
\begin{equation*}
    \mathbb{E}\left[\max _{0 \leq k \leq N} \left\|X_{t_k}- X_k^h\right\|^2\right] \leq C h
\end{equation*}
\begin{proof}
    The proof needs following steps.

    \vspace{0.5em}
    \textbf{Step 1.} SDE on Euclidean: For $\pi \colon V \sto M$, by Corollary \ref{cor:proj_by_pi}, $\Pi(x) \defeq D\pi(x) \colon \R^n \sto \R^n$ for $x \in M$ is the orthogonal projection onto $T_xM$. Then we can choose
    \begin{equation*}
        dB^M_t = \Pi \circ dW_t,
    \end{equation*}
    where $W$ is a standard $\R^n$-Brownian motion that is also independent of $X_0$. Then
    \begin{equation}\label{eq:eucl_ext_diff}
        dX_t = v(X_t)dt + \Pi(X_t) \circ dW_t = \tilde{a}(X_t) dt + \Pi(X_t)dW_t,
    \end{equation}
    as an Euclidean SDE, where
    \begin{equation*}
        \tilde{a}(x) = v(x) + A(x),\quad A^j(x) = \frac{1}{2} \sum_{\alpha = 1}^n \Pi_{\alpha}(x)\bc{\Pi_{j\alpha}(x)},
    \end{equation*}
    for $\Pi(x) = (\Pi_{j\alpha}(x))$ when fixing the standard basis $\bb{e_1,\cdots, e_n}$ in $\R^n$. Moreover, for
    \begin{equation*}
        \Pi_\alpha(x) = \Pi(x)e_\alpha \in T_x M,
    \end{equation*}
    we have
    \begin{equation*}
        \Pi_{j\alpha}(x) = \Pi_\alpha (x^j)(x),
    \end{equation*}
    where $x^j \colon M \sto \R$ is the $j$-th component of the canonical embedding $x \colon M \hookrightarrow \R^n$.
    
    \vspace{0.25em}
    \textbf{Claim:} If $\bb{E_i(x)}_{i=1}^d$ is an orthonormal frame of $T_xM \subset \R^n$, then we have
    \begin{equation*}
        A(x) = \frac{1}{2} \sum_{i=1}^d \mathrm{II}_x(E_i(x),E_i(x)).
    \end{equation*}
    To prove that, first, by
    \begin{equation*}
        \Delta_Mx^j = \sum_{\alpha = 1}^N\Pi_\alpha (\Pi_\alpha(x^j)) = \sum_{\alpha = 1}^N\Pi_\alpha (\Pi_{j\alpha}),
    \end{equation*}
    we have
    \begin{equation*}
        A^j(x) = \frac{1}{2}\Delta_Mx^j(x).
    \end{equation*}
    On the other hand, we know
    \begin{equation*}
        \Delta_Mx^j(x) = \sum_{i=1}^d\mathrm{II}^j_x(E_i(x),E_i(x)).
    \end{equation*}

    \vspace{0.5em}
    \textbf{Step 2.} Extension on $\R^n$: Because $v,A$ and $\Pi$ are only make sense on $M$, we cannot deal with (\ref{eq:eucl_ext_diff}) as the usual SDE on $\R^n$. So we need to extend it on $\R^n$ with well-posed properties. First, because $M$ is closed, it can choose a smaller open $U$ such that $M \subset U \subset V$. Then it can choose a bump function $\chi \colon \R^n \sto [0,1]$ such that $\chi|_{\bar{U}} = 1$, $\chi_{\R^n \backslash V} = 0$, and $\norm{D\chi} \leq C_\chi < \infty$. 
    \begin{itemize}
        \item Extending $\Pi$: Let
        \begin{equation*}
            \tilde{\Pi}(y) = \chi(y) \Pi(\pi(y)),\quad \forall ~ y \in \R^n.
        \end{equation*}
        Next, we need to guarantee the global Lipschitz continuity of $\tilde{\Pi}$. By Lemma \ref{lem:bound_secform_boun_proj} with Assumption \ref{assum:conv_gem_1},
        \begin{equation*}
            \sup_{x \in M}\norm{D\Pi(x)[v]}_{\op{op}} = \sup_{x\in M} \norm{\clo{\nabla}_v\Pi(x)}_{\op{op}} < C_\Pi\norm{v},\quad \forall~x \in M,~\forall v \in T_xM.
        \end{equation*}
        Then, for any $y \in \R^n$ and $w \in \R^n$, we have
        \begin{align*}
            \norm{D\tilde{\Pi}(y)[w]} &= \norm{\Pi(\pi(y))D\chi(y)[w] + \chi(y) D \Pi(\pi(y))[D\pi(y)[w]] } \\
            &\leq C_\chi \norm{w} + C_\Pi C_\pi\norm{w},
        \end{align*}
        because $D\pi(y)[w] \in T_{\pi(y)}M$ and by Assumption \ref{assum:conv_gem_2}, $\norm{D\pi(y)} \leq C_\pi < \infty$. Therefore,
        \begin{equation*}
            \sup_{y \in \R^n} \norm{D\tilde{\Pi}(y)}_{\op{op}} \leq L_\Pi \defeq C_\chi + C_\Pi C_\pi,
        \end{equation*}
        i.e., $\tilde{\Pi}$ is globally Lipschitz continuous,
        \begin{equation*}
            \norm{\tilde{\Pi}(y_1) - \tilde{\Pi}(y_2)}_{\op{op}} \leq L_\Pi \norm{y_1 - y_2},\quad \forall~y_1,y_2 \in \R^n.
        \end{equation*}
        In the following, we denote $\tilde{\Pi}$ by $\Pi$ for convenience.

        \item Extending $v$: For $v$ satisfying Assumption \ref{assum:conv_gem_3}, by Corollary \ref{cor:bound_ext_deriv_vec},
        \begin{equation*}
            \sup_{x \in M} \norm{Dv(x)[u]} =\sup_{x \in M} \norm{\clo{\nabla}_uv(x)} < C_v\norm{u},\quad \forall~u \in T_xM.
        \end{equation*}
        Similarly, let
        \begin{equation*}
            \tilde{v}(y) = \chi(y) v(\pi(y)),\quad \forall ~ y \in \R^n.
        \end{equation*}
        Then for any $y \in \R^n$ and $w \in \R^n$,
        \begin{align*}
            \norm{D\tilde{v}(y)[w]} &= \norm{v(\pi(y))D\chi(y)[w] + \chi(y) D v(\pi(y))[D\pi(y)[w]] } \\
            &\leq C_\chi C_v^\prime \norm{w} + C_v C_\pi\norm{w},
        \end{align*}
        because $\sup_{x \in M} \norm{v(x)} \leq C_v^\prime$ of Assumption \ref{assum:conv_gem_3}. It follows that $\tilde{v}$ is globally Lipschitz continuous,
        \begin{equation*}
            \sup_{y\in \R^n} \norm{D\tilde{v}(y)}_{\op{op}} \leq L_v \defeq C_\chi C_v^\prime + C_v C_\pi,
        \end{equation*}
        i.e.,
        \begin{equation*}
            \norm{\tilde{v}(y_1) - \tilde{v}(y_2)} \leq L_v \norm{y_1 - y_2},\quad \forall ~ y_1,y_2 \in \R^n.
        \end{equation*}
        We also do not distinguish $v$ and $\tilde{v}$.

        \item Extending $A$: by above, we have known
        \begin{equation*}
            A(x) = \sum_{i = 1}^d \mathrm{II}_x(E_i(x),E_i(x)) \in N_xM,\quad \forall~ x \in M.
        \end{equation*}
        Moreover, we let $\bb{E_1,\cdots,E_d}$ be geodesic at $x$, i.e., $\nabla_{E_i}E_j(x) = 0$. Then from the proof of Lemma \ref{lem:taylo_exp_bounded} with Assumption \ref{assum:conv_gem_1}, and by $\bb{E_1,\cdots,E_d}$ geodesic, i.e., $\nabla_vE_i = 0$, we have
        \begin{equation*}
            \sup_x \norm{DA(x)[u]} = \sup_{x \in M} \norm{\clo{\nabla}_u A(x)} \leq C_A \norm{u},\quad \forall~ u\in T_xM.
        \end{equation*}
        Let
        \begin{equation*}
            \tilde{A}(y) = \chi(y) A(\pi(y)),\quad \forall ~ y \in \R^n.
        \end{equation*}
        Then for any $y \in \R^n$ and $w \in \R^n$,
        \begin{align*}
            \norm{\tilde{A}(y)[w]} &= \norm{A(\pi(y))D\chi(y)[w] + \chi(y) D A(\pi(y))[D\pi(y)[w]]} \\
            &\leq C_{\mathrm{II}}C_\chi \norm{w} + C_AC_\pi \norm{w}
        \end{align*}
        Therefore, $\tilde{A}$ is globally Lipschitz continuous. In the following, we also denote $\tilde{A}$ by $A$.
    \end{itemize}
    Then for such extended $v,A$, and $\Pi$, we get SDE (\ref{eq:eucl_ext_diff}) defined on Euclidean space with globally Lipschitz continuous coefficients.

    \vspace{0.5em}
    \textbf{Step 3.} Euclidean EM discretization: We can consider the Euclidean EM algorithm. Let $t_k=k h, N h=T, \Delta W_k=W_{t_{k+1}}-W_{t_k}$, and define
    \begin{equation*}
        Y_{k+1}^h=Y_k^h+\tilde{a}\left(Y_k^h\right) h+\Pi\left(Y_k^h\right) \Delta W_k, \quad Y_0^h=X_0.
    \end{equation*}
    By the global Lipschitz continuity of $\tilde{a}$, and Assumption \ref{assum:conv_gem_4} of $X_0 \in L^2$, we have
    \begin{equation*}
        \mathbb{E}\left[\max _{0 \leq k \leq N}\left\|X_{t_k}-Y_k^h\right\|^2\right] \leq C_T h.
    \end{equation*}
    So the next goal is to bound
    \begin{equation*}
        \mathbb{E}\left[\max _{0 \leq k \leq N}\left\|X_{k}^h-Y_k^h\right\|^2\right].
    \end{equation*}
    Moreover, let
    \begin{equation*}
        \Phi^E_h(x,z) = x + \tilde{a}(x)h + \Pi(x)z.
    \end{equation*}
    So
    \begin{equation*}
        Y_{k+1}^h = \Phi^E_h(Y_k^h,\Delta W_k)
    \end{equation*}
    
     \vspace{0.5em}
    \textbf{Step 4.} Geometric EM discretization: For
    \begin{equation*}
        X_{k+1}^h=\exp _{X_k^h}\left(h v\left(X_k^h\right)+\sqrt{h} \xi_{k+1}\right)
    \end{equation*}
    with $\xi_{k+1}$ a standard Gaussian on $T_xM$. WTLG, assume
    \begin{equation*}
        \xi_{k+1}:=\frac{1}{\sqrt{h}} \Pi\left(X_k^h\right) \Delta W_k,\quad \Delta W_k = W_{t_{k+1}} - W_{t_k},
    \end{equation*}
    where $W$ is the Brownian motion in (\ref{eq:eucl_ext_diff}). So the geometric EM is
    \begin{equation*}
        X_{k+1}^h=\exp _{X_k^h}\left(h v\left(X_k^h\right)+\Pi\left(X_k^h\right) \Delta W_k\right).
    \end{equation*}
    Let
    \begin{equation*}
        \Phi^G_h(x,z) = \exp_x(hv(x) + \Pi(x)z),
    \end{equation*}
    i.e., we have
    \begin{equation*}
        X_{k+1}^h= \Phi^G_h(X_k^h,\Delta W_k).
    \end{equation*}

    Moreover, by Lemma \ref{lem:taylo_exp_bounded}, for $u = hv(x) + \Pi(x)z$,
    \begin{equation*}
        \Phi^G_h(x,z) = x + hv(x) + \Pi(x)z + \frac{1}{2}\mathrm{II}_x(u,u) + R_3(x,u),
    \end{equation*}
    where
    \begin{align*}
        \norm{R_3(x,u)} \leq C_1\norm{u}^3,
    \end{align*}
    where $C_1$ is independent of $x$.

    \vspace{0.25em}
    \textbf{Claim:} For $z \sim \mathcal{N}(0,hI_n)$, we have
    \begin{equation*}
        \E\bj{\mathrm{II}_x(\Pi(x)z,\Pi(x)z)} = 2hA(x).
    \end{equation*}
    For
    \begin{equation*}
        \Pi(x)z = \sum_{i=1}^d \zeta_i E_i(x),\quad \zeta_i = \inn{z,E_i(x)},
    \end{equation*}
    then $\zeta = (\zeta_1,\cdots,\zeta_d)$ is a Gaussian with
    \begin{equation*}
        \mathbb{E}\left[\zeta_i \zeta_j\right]=\mathbb{E}\left[\left\langle z, E_i\right\rangle\left\langle z, E_j\right\rangle\right]=E_i^{\top} \mathbb{E}\left[z z^{\top}\right] E_j=E_i^{\top}\left(h I_N\right) E_j=h \delta_{i j},
    \end{equation*}
    i.e., $\zeta \sim \mathcal{N}(0,hI_d)$. Because $\mathrm{II}_x$ is bilinear,
    \begin{equation}\label{eq:bound_2ndform_guass}
        \mathrm{II}_x(\Pi(x)z,\Pi(x)z) = \sum_{i,j=1}^d \zeta_i\zeta_j \mathrm{II}_x(E_i,E_j).
    \end{equation}
    Therefore,
    \begin{align*}
        \E\bj{\mathrm{II}_x(\Pi(x)z,\Pi(x)z)} &= \sum_{i,j=1}^d \E\bj{\zeta_i\zeta_j} \mathrm{II}_x(E_i,E_j) \\
        &= h \sum_{i=1}^d \mathrm{II}_x(E_i,E_i)= 2hA(x).
    \end{align*}

    Therefore, for $u = hv(x) + \Pi(x)z$ and $z = \Delta W_k \sim \mathcal{N}(0,hI_n)$,
    \begin{equation*}
        \E\bj{\mathrm{II}_x(u,u)} = \E\bj{\mathrm{II}_x(\Pi(x)z,\Pi(x)z)} + h^2\mathrm{II}_x(v,v) = 2hA(x) + \mathcal{O}(h^2).
    \end{equation*}
    It follows that
    \begin{align*}
        \mathbb{E}\left[\Phi_h^G(x, \Delta W_k)\right] & =x+ h v(x)+h A(x)+\mathcal{O}(h^2) \\
        & =x + \tilde{a}(x) h+\mathcal{O}(h^2).
    \end{align*}
    Note that by (\ref{eq:bound_2ndform_guass}) and Assumption \ref{assum:conv_gem_1}, we also have
    \begin{equation*}
        \E\bj{\norm{\mathrm{II}_x(\Pi(x)z,\Pi(x)z)}^2} = \mathcal{O}(h^2).
    \end{equation*}

    \vspace{0.5em}
    \textbf{Step 5.} Local difference: For $\Phi^E_h, \Phi^G_h$ defined as above, we have
    \begin{equation*}
        \Phi_h^G(x, z)-\Phi_h^E(x, z) = -hA(x) + \frac{1}{2}\Pi_x(u,u) + R_3(x,u).
    \end{equation*}
    Let
    \begin{equation*}
        \Delta_h(x) = \Phi_h^G(x, \Delta W_k)-\Phi_h^E(x, \Delta W_k) = Q_h(x) + R_3(x,u),
    \end{equation*}
    where
    \begin{equation*}
        Q_h(x) \defeq \frac{1}{2}\Pi_x(u,u) - hA(x),\quad u = hv(x) + \Pi(x)\Delta W_k.
    \end{equation*}
    Writing
    \begin{align*}
        Q_h(x) & =\frac{1}{2} \mathrm{II}_x(u, u)-h A(x) \\
        & =\bc{\frac{1}{2} \mathrm{II}_x(\Pi \Delta W_k, \Pi \Delta W_k)-h A(x)}+\frac{1}{2} \mathrm{II}_x(h v, h v)+\mathrm{II}_x(h v, \Pi \Delta W_k) \\
        & = T_1 + T_2 + T_3.
    \end{align*}
    Before continuing, let's recall that for $u = hv(x) + \Pi(x)\Delta W_k$, by the boundedness of $v$ and $\E[\norm{\Delta W_k}^p] = \mathcal{O}(h^{p/2})$, we have
    \begin{equation*}
        \mathbb{E}\bj{\norm{u}^3}=\mathcal{O}(h^{3/2}), \quad \mathbb{E}\bj{\norm{u}^4}=\mathcal{O}\left(h^2\right), \quad \mathbb{E}\bj{\norm{u}^6}=\mathcal{O}\left(h^3\right) .
    \end{equation*}
    \begin{itemize}
        \item First momentum: For $T_1$, by above, we have $\E[T_1] = 0$. It is obvious that $\norm{\E[T_2]} = \mathcal{O}(h^2)$ and $\E[T_3] = 0$. So 
        \begin{equation*}
            \norm{\E\bj{Q_h(x)}} = \mathcal{O}(h^2).
        \end{equation*}
        For $R_3$,
        \begin{equation*}
            \norm{R_3} \leq C_1\norm{u}^3~\Rightarrow~ \norm{\E[R_3]} \leq \E\bj{\norm{R_3}} \leq C_1 \E\bj{\norm{u}^3} = \mathcal{O}(h^{3/2}).
        \end{equation*}
        Therefore,
        \begin{equation*}
            \norm{\E\bj{\Delta_h(x)}} \leq  \norm{\E\bj{Q_h(x)}} + \norm{\E\bj{R_3}} \leq \mathcal{O}(h^{3/2}).
        \end{equation*}

        \item Second momentum: By above, because of the boundedness of second fundamental, we have
        \begin{equation*}
            \E\bj{\norm{T_1}^2} =\mathcal{O}(h^2),\quad  \E\bj{\norm{T_2}^2} =\mathcal{O}(h^4),\quad  \E\bj{\norm{T_3}^2} =\mathcal{O}(h^3),
        \end{equation*}
        which implies that
        \begin{equation*}
            \E\bj{\norm{Q_h(x)}^2} \leq \mathcal{O}(h^2).
        \end{equation*}
        And
        \begin{equation*}
            \norm{R_3} \leq C_1\norm{u}^3~\Rightarrow~ \E\bj{\norm{R_3}^2} \leq C^2_1\E\bj{\norm{u}^6} = \mathcal{O}(h^3).
        \end{equation*}
        Therefore,
        \begin{equation*}
            \E\bj{\norm{\Delta_h(x)}^2} = \mathcal{O}(h^2)
        \end{equation*}
    \end{itemize}   

    \vspace{0.5em}
    \textbf{Step 6.} Global difference: Let $e_k = X^h_k - Y^h_k$. Note that
    \begin{equation*}
        X_{k+1}^h=\Phi_h^G\left(X_k^h, \Delta W_k\right), \quad Y_{k+1}^h=\Phi_h^E\left(Y_k^h, \Delta W_k\right),
    \end{equation*}
    where $\Delta W_k = W_{t_{k+1}} - W_{t_k}$. Therefore,
    \begin{align*}
        e_{k+1} &= \Phi_h^G\left(X_k^h, \Delta W_k\right) - \Phi_h^E\left(Y_k^h, \Delta W_k\right) \\
        &= \bc{\Phi_h^E\left(X_k^h, \Delta W_k\right) - \Phi_h^E\left(Y_k^h, \Delta W_k\right)} + \bc{\Phi_h^G\left(X_k^h, \Delta W_k\right) - \Phi_h^E\left(X_k^h, \Delta W_k\right)} \\
        &= G_k + D_k,
    \end{align*}
    where
    \begin{equation*}
        G_k = \Phi_h^E\left(X_k^h, \Delta W_k\right) - \Phi_h^E\left(Y_k^h, \Delta W_k\right),\quad D_k = \Delta_h(X_k^h).
    \end{equation*}
    Let $\mathbb{F}_t = \sigma(X_0,W_s \colon s\leq t)$. Therefore, $X^h_k,Y^h_k$ are $\mathcal{F}_{t_k} \eqdef \mathcal{F}_k$-measurable. Moreover, because $W$ is independent of $X_0$, $W$ is also a Brownian motion w.s.t. $(\mathcal{F}_t)$. So
    \begin{equation*}
        \E\bj{\norm{e_{k+1}}^2 \mid \mathcal{F}_k} = \E\bj{\norm{G_k}^2 \mid \mathcal{F}_k} + \E\bj{ \inn{G_k,D_k} \mid \mathcal{F}_k} + \E\bj{\norm{D_k}^2 \mid \mathcal{F}_k}.
    \end{equation*}
    For these three terms, let's analyze them separably.
    \begin{itemize}
        \item For $G_k$, first note that
        \begin{equation*}
            G_k = e_k + h(\tilde{a}(X^h_k) - \tilde{a}(Y^h_k)) + (\Pi(X^h_k) - \Pi(Y^h_k))\Delta W_k.
        \end{equation*}
        So we have
        \begin{equation*}
            \E\bj{G_k \mid \mathcal{F}_k} = e_k + h(\tilde{a}(X^h_k) - \tilde{a}(Y^h_k)).
        \end{equation*}
        Then by global Lipschitz continuity of $\tilde{a}$, we have
        \begin{equation*}
            \norm{\E\bj{G_k \mid \mathcal{F}_k}} \leq (1 + L_1h)\norm{e_k}.
        \end{equation*}
        On the other hand, by the global Lipschitz continuity of $\Pi$,
        \begin{equation*}
            G_k - \E\bj{G_k \mid \mathcal{F}_k} =  (\Pi(X^h_k) - \Pi(Y^h_k))\Delta W_k~\Rightarrow~ \E\bj{\norm{G_k - \E\bj{G_k \mid \mathcal{F}_k}}^2 \mid \mathcal{F}_k} \leq L_2h \norm{e_k}^2.
        \end{equation*}
        Therefore, 
        \begin{align*}
            \E\bj{\norm{G_k}^2 \mid \mathcal{F}_k} &= \norm{\E\bj{G_k \mid \mathcal{F}_k}}^2 + \E\bj{\norm{G_k - \E\bj{G_k \mid \mathcal{F}_k}}^2 \mid \mathcal{F}_k} \\
            &\leq (1 + C_2h) \norm{e_k}^2
        \end{align*}

        \item For $D_k$,
        \begin{equation*}
            \mu_k \defeq \E\bj{D_k \mid \mathcal{F}_k} = \E\bj{\Delta_h(x)}_{x = X^h_k}.
        \end{equation*}
        Therefore, by above step, we have
        \begin{equation*}
            \norm{\mu_k} \leq C_3h^{3/2},
        \end{equation*}
        and
        \begin{equation*}
            \E\bj{\norm{D_k}^2 \mid \mathcal{F}_k} = \E\bj{\norm{\Delta_h(x)}^2}_{x = X^h_k} \leq \mathcal{O}(h^2).
        \end{equation*}
        Moreover, let $\widetilde{D}_k \defeq D_k - \mu_k$, so $\E\bj{\widetilde{D}_k \mid \mathcal{F}_k} = 0$ and
        \begin{equation*}
            \E\bj{\|\widetilde{D}_k\|^2 \mid \mathcal{F}_k} \leq C_4 h^2
        \end{equation*}

        \item For the cross term, by $D_k = \mu_k + \widetilde{D}_k$ and $\mu_k$ $\mathcal{F}_k$-measurable,
        \begin{equation*}
            \mathbb{E}\left[\left\langle G_k, D_k\right\rangle \mid \mathcal{F}_k\right]=\left\langle\mathbb{E}\left[G_k \mid \mathcal{F}_k\right], \mu_k\right\rangle+\mathbb{E}\left[\left\langle G_k, \widetilde{D}_k\right\rangle \mid \mathcal{F}_k\right] .
        \end{equation*}
        For the first term, by above, we have
        \begin{equation*}
            \abs{\left\langle\mathbb{E}\left[G_k \mid \mathcal{F}_k\right], \mu_k\right\rangle} \leq \norm{\mathbb{E}\left[G_k \mid \mathcal{F}_k\right]}\norm{\mu_k} \leq C_3(1+L_1h)h^{3/2}\norm{e_k}.
        \end{equation*}
        Moreover, by $2ab \leq \varepsilon a^2 + (1/\varepsilon)b^2$ and setting $\varepsilon=h$,
        \begin{equation*}
            h^{3/2}\norm{e_k} \leq \frac{1}{2}\bc{h\norm{e_k}^2 + h^2}.
        \end{equation*}
        So
        \begin{equation*}
            \abs{\left\langle\mathbb{E}\left[G_k \mid \mathcal{F}_k\right], \mu_k\right\rangle} \leq C_5h \norm{e_k}^2 +  \mathcal{O}(h^2).
        \end{equation*}
        For the second term, first, because $\E[\widetilde{D}_k \mid \mathcal{F}_k] = 0$,
        \begin{equation*}
            \mathbb{E}\left[\left\langle G_k, \widetilde{D}_k\right\rangle \mid \mathcal{F}_k\right]=\mathbb{E}\left[\left\langle G_k-\mathbb{E}\left[G_k \mid \mathcal{F}_k\right], \widetilde{D}_k\right\rangle \mid \mathcal{F}_k\right].
        \end{equation*}
        Then by the conditional CBS inequality,
        \begin{equation*}
            \left|\mathbb{E}\left[\left\langle G_k, \widetilde{D}_k\right\rangle \mid \mathcal{F}_k\right]\right| \leq \sqrt{\mathbb{E}\left[\left\|G_k-\mathbb{E}\left[G_k \mid \mathcal{F}_k\right]\right\|^2 \mid \mathcal{F}_k\right]} \sqrt{\mathbb{E}[\|\widetilde{D}_k\|^2 \mid \mathcal{F}_k]}.
        \end{equation*}
        By above, 
        \begin{equation*}
            \mathbb{E}\left[\left\|G_k-\mathbb{E}\left[G_k \mid \mathcal{F}_k\right]\right\|^2 \mid \mathcal{F}_k\right] \leq L_2 h \norm{e_k}^2,\quad \mathbb{E}[\|\widetilde{D}_k\|^2 \mid \mathcal{F}_k]\leq C_4h^2.
        \end{equation*}
        So
        \begin{equation*}
            \left|\mathbb{E}\left[\left\langle G_k, \widetilde{D}_k\right\rangle \mid \mathcal{F}_k\right]\right| \leq \sqrt{L_2C_4} h^{3/2}\norm{e_k} \leq C_6h \norm{e_k}^2 + \mathcal{O}(h^2).
        \end{equation*}
        Therefore,
        \begin{equation*}
            \abs{\mathbb{E}\left[\left\langle G_k, D_k\right\rangle \mid \mathcal{F}_k\right]} \leq (C_5+C_6)h \norm{e_k}^2 + C_7h^2.
        \end{equation*}
    \end{itemize}
    Therefore, we have
    \begin{equation*}
        \E\bj{\norm{e_{k+1}}^2 \mid \mathcal{F}_k} \leq (1 + C_\alpha h)\norm{e_k}^2 + C_\beta h^2,
    \end{equation*}
    which implies that
    \begin{equation*}
        \E\bj{\norm{e_{k+1}}^2} \leq (1 + C_\alpha h)\E\bj{\norm{e_k}^2} + C_\beta h^2.
    \end{equation*}
    Then by the discrete Gr\"onwall's inequality,
    \begin{equation*}
        \E\bj{\max_{0\leq k \leq N} \norm{X_k^h-Y_k^h}^2} \leq C_T^\prime h.
    \end{equation*}    

    \vspace{0.5em}
    \textbf{Step 5.} Final step: Combining all, 
    \begin{align*}
        \E\bj{\max_{0\leq k \leq N} \norm{X_{t_k}-Y_k^h}^2} &\leq 2\E\bj{\max_{0\leq k \leq N} \norm{X_{t_k}-Y_k^h}^2} + 2\E\bj{\max_{0\leq k \leq N} \norm{X_k^h-Y_k^h}^2} \\
        &\leq Ch. \qedhere
    \end{align*}
\end{proof}

\begin{lem}\label{lem:diff_of_pi}
    Let $M \subset \R^n$ be a Riemannian submanifold. Let $V \subset \R^n$ be a tubular neighborhood of $M$ with the canonical projection $\pi \colon V \sto M \subset \R^n$. Then for any $y \in V$, $D\pi(y) \colon \R^n \sto \R^n$ with
    \begin{equation*}
        \ker D\pi(y) = N_{\pi(y)} M,\quad \Img D\pi(y) = T_{\pi(y)}M.
    \end{equation*}
\end{lem}
\begin{proof}
    First, we know that for any $x \in M$,
    \begin{equation*}
        \pi^{-1}(x) = \bb{x+v \colon \norm{v} \text{ is small. }} = (x + N_xM) \cap V.
    \end{equation*}

    For any $y \in V$, let $x = \pi(y)$, i.e., $y \in \pi^{-1}(x)$.
    \begin{itemize}
        \item $N_xM \subset \ker D\pi(y)$: Let $w \in N_xM$ and choose $\gamma(t) = y + tw$. Since $y \in \pi^{-1}(x) = (x + N_xM) \cap V$,
        \begin{equation*}
            \gamma(t) = y + tw = x + (v + tw) \in x + N_xM,
        \end{equation*}
        which implies that $\pi(\gamma(t)) \equiv x$. Then
        \begin{equation*}
            D\pi(y)[w] = \lv{\frac{d}{dt}}_{t = 0}\pi(\gamma(t)) = 0.
        \end{equation*}
        So $N_xM \subset \ker D\pi(y)$.

        \item $T_xM = \Img D\pi(y)$: First, because $\pi \colon V \sto M$, it obviously $\Img D\pi(y) \subset T_xM$. Conversely, let $F \colon U \sto V$ be the tubular neighborhood diffeomorphism, where
        \begin{equation*}
            U = \bb{(x,v) \colon x\in M,\quad v \in N_xM \text{ with } \norm{v} \leq \varepsilon(x)} \subset NM,
        \end{equation*}
        i.e, $F(x,v) = x+v$. Let $pr \colon U \sto M$ by $pr(x,v) = x$. Then $\pi = pr \circ F^{-1} \colon V \sto M$. So
        \begin{equation*}
            D\pi(y) = Dpr(x,v) \circ D(F^{-1})(y)
        \end{equation*}
        Note that $D(F^{-1})(y)$ is an isomorphism. So $\op{rank} D\pi(y) = \op{rank} Dpr(x,v)$. For $pr$, note that
        \begin{equation*}
            Dpr(x,v) \colon T_{(x,v)}(NM) \cong T_xM \otimes N_xM \sto T_xM
        \end{equation*}
        is a surjection. So
        \begin{equation*}
            \op{rank} D\pi(y) = \op{rank} Dpr(x,v) = \op{dim} (T_xM),
        \end{equation*}
        which implies that
        \begin{equation*}
            T_xM = \Img D\pi(y).
        \end{equation*}

        \item $N_xM = \ker D\pi(y)$: It is just by
        \begin{equation*}
            \op{dim}(\ker D\pi(y)) = n - \op{dim} D\pi(y) = n - \op{dim} (T_xM) = \op{dim} (N_xM). \qedhere
        \end{equation*}
    \end{itemize}
\end{proof}


\begin{cor}\label{cor:proj_by_pi}
    Let $M \subset \R^n$ be a Riemannian submanifold. Let $V \subset \R^n$ be a tubular neighborhood of $M$ with the canonical projection $\pi \colon V \sto M \subset \R^n$. Then for any $x \in M$, $D\pi(x) \colon \R^n \sto \R^n$ is the orthogonal projection from $\R^n$ onto $T_xM$.
\end{cor}
\begin{proof}
    Fix $x \in M$. For any $v \in T_xM \subset \R^n$, let $c \colon (-\varepsilon,\varepsilon) \sto M$ be a curve with $c(0) = x$ and $\dot{c}(0) = v$. Because $\pi |_M = \op{id}_M$, $\pi(c(t)) = c(t)$. It follows that
    \begin{equation*}
        D\pi(x)[v] = \lv{\frac{d}{dt}}_{t = 0} \pi(c(t)) = \lv{\frac{d}{dt}}_{t = 0} c(t) = v.
    \end{equation*}
    So $D\pi(x) \colon \R^n = T_xM \otimes N_xM \sto \R^n$ is identity on $T_xM$. By Lemma \ref{lem:diff_of_pi}, $D\pi(x)$ is an orthogonal projection.
\end{proof}


\begin{lem}\label{lem:bound_secform_boun_proj}
    Assume $M \subset \R^n$ properly and isometrically embedded and assume the second fundamental form is uniformly bounded, i.e.,
    \begin{equation*}
        \sup_{x \in M} \norm{\mathrm{II}_x}_{\op{op}}< \infty.
    \end{equation*}
    Then we have
    \begin{equation*}
       \sup_{x \in M}\norm{\nabla \Pi(x)}_{\op{op}} < \infty ,\quad \sup_{x \in M}\sup_{v \in T_xM, \norm{v} = 1}\norm{\clo{\nabla}_v\Pi(x)}_{\op{op}} < \infty
    \end{equation*}
\end{lem}
\begin{proof}
    We need to prove for the following separated steps.
    \begin{itemize}
        \item \textbf{Shape operator:} For any $\eta \in \Gamma(NM)$, consider the shape operator $s_\eta \colon \Gamma(TM) \sto \Gamma(TM)$ as
        \begin{equation*}
            \inn{s_\eta(X),Y} = \inn{\mathrm{II}(X,Y),\eta},\quad X,Y \in \Gamma(TM).
        \end{equation*}
        Because the second fundamental form is uniformly bounded, i.e.,
        \begin{equation*}
            \sup_{x \in M} \sup_{\norm{X},\norm{Y} = 1} \norm{\mathrm{II}_x(X,Y)} \leq C < \infty,
        \end{equation*}
        we have
        \begin{equation*}
            \norm{s_\eta}_{\op{op}} \leq C\norm{\eta},
        \end{equation*}

        \item \textbf{Intrinsic Derivative:} For any $w \in \R^N$, $\Pi(x)w \in T_xM$, and by smoothness, we have $\Pi w \in \Gamma(TM)$, i.e, $\Pi \in \Gamma(M,\op{Hom}(\R^N,TM))$. Therefore, we define $\nabla_X\Pi \in \Gamma(M,\op{Hom}(\R^N,TM))$ as
        \begin{equation*}
            (\nabla_X\Pi)(x)w \defeq \nabla_X\bc{\Pi(x)w},\quad \forall~ x \in M.
        \end{equation*}
        Note that for $w \in \R^N$,
        \begin{equation*}
            w = w(x)^\top + w(x)^\perp,
        \end{equation*}
        where
        \begin{equation*}
            w(x)^\top = \Pi(x)w \in T_xM,\quad w(x)^\perp = w - w(x)^\top \in N_xM.
        \end{equation*}
        Because $w$ is constant on $\R^N$,
        \begin{equation*}
            0 = \clo{\nabla}_X w =  \clo{\nabla}_X w^\top  + \clo{\nabla}_X w^\perp.
        \end{equation*}
        For $\clo{\nabla}_X w^\top$, we know
        \begin{equation*}
            \clo{\nabla}_X w^\top = \nabla_X w^\top + \mathrm{II}(w^\top,X).
        \end{equation*}
        For $w^\perp \in \Gamma(NM)$, by Weingarten equation $(\clo{\nabla}_X w^\perp)^\top = - s_{w^\perp}(X)$, we have
        \begin{equation*}
            \clo{\nabla}_X w^\perp = -s_{w^\perp}(X) + \nabla_X^\perp w^\perp.
        \end{equation*}
        Therefore, 
        \begin{equation*}
            \nabla_X w^\top -s_{w^\perp}(X) = - \bc{\nabla_X^\perp w^\perp + \mathrm{II}(w^\top,X)}.
        \end{equation*}
        Note that RHS $\in \Gamma(TM)$ and LHS $\in \Gamma(NM)$. So they are both $0$, i.e.,
        \begin{equation*}
            (\nabla_X\Pi)w= \nabla_X\bc{\Pi w} = \nabla_X w^\top = s_{w^\perp}(X).
        \end{equation*}
        By the boundedness of $s$, we have
        \begin{equation*}
            \norm{(\nabla_X\Pi)(x)w} = \norm{s_{w^\perp(x)}(X)} \leq \norm{s_{w^\perp(x)}}_{\op{op}}\norm{X} \leq C\norm{w^\perp(x)}\norm{X},
        \end{equation*}
        which implies that
        \begin{equation*}
            \norm{\nabla_X \Pi(x)}_{\op{op}} = \sup_{\norm{w} = 1}\norm{(\nabla_X\Pi)(x)w} \leq C\norm{X}~\Rightarrow~\norm{\nabla \Pi(x)}_{\op{op}} < C
        \end{equation*}
        Note that $C$ is independent of $x \in M$.

        \item \textbf{Extrinsic Derivative:} For any $x\in M$, $v \in T_xM$ with $\norm{v} = 1$, and $w \in \R^n$ with $\norm{w} = 1$,
        \begin{equation*}
            \clo{\nabla}_{v}(\Pi(x)w) = \nabla_{v}(\Pi(x)w) + \mathrm{II}(\Pi(x)w,v).
        \end{equation*}
        So
        \begin{equation*}
            \norm{\clo{\nabla}_{v}(\Pi(x)w)} \leq \norm{\nabla_{v}(\Pi(x)w)} + \norm{\mathrm{II}(\Pi(x)w,v)} \leq 2C,
        \end{equation*}
        which implies that
        \begin{equation*}
            \sup_{x \in M}\sup_{v \in T_xM, \norm{v} = 1}\norm{\clo{\nabla}_v\Pi(x)}_{\op{op}} < \infty. \qedhere
        \end{equation*}
    \end{itemize}
\end{proof}



\begin{cor}\label{cor:lipsch_intrin_proj}
    Under same assumptions as Lemma \ref{lem:bound_secform_boun_proj}, the orthogonal projection $\Pi$ is globally Lipschitz continuous w.s.t. $d_M$ on $M$, i.e., there exists some $L > 0$ such that
    \begin{equation*}
        \norm{\Pi(x) - \Pi(y)}_{\op{op}} \leq L d_M(x,y),\quad \forall~x,y \in M
    \end{equation*}
\end{cor}
\begin{proof}
    For any $x,y \in M$, by the completeness, let $\gamma \colon [0,\ell] \sto M$ be the minimizing unit geodesic from $x$ to $y$, i.e.,
    \begin{equation*}
        \gamma(0) = x,\quad \gamma(\ell) = y,\quad\norm{\dot{\gamma}(t)} \equiv 1,
    \end{equation*}
    and $d_M(x,y) = \ell$. Let $F(t) \defeq \Pi(\gamma(t))$. When viewing $\Pi \colon \R^N \sto \R^N$ as a matrix and $\gamma(t) \in \R^N$
    \begin{equation*}
        \frac{d}{dt}F(t) = \clo{\nabla}_{\dot{\gamma}(t)}\Pi(\gamma(t)).
    \end{equation*}
    By Lemma \ref{lem:bound_secform_boun_proj},
    \begin{equation*}
        \norm{\frac{d}{dt}F(t)}_{\op{op}} = \norm{\clo{\nabla}_{\dot{\gamma}(t)}\Pi(\gamma(t))}_{\op{op}} \leq 2C.
    \end{equation*}
    It implies that
    \begin{align*}
        \norm{\Pi(x) - \Pi(y)}_{\op{op}} &= \norm{F(\ell) - F(0)}_{\op{op}} \\
        &= \norm{\int_0^\ell  \frac{d}{dt}F(t) dt}_{\op{op}} \leq \int_0^\ell  \norm{\frac{d}{dt}F(t)}_{\op{op}} dt \leq 2C d_M(x,y). \qedhere
    \end{align*}
\end{proof}

\begin{cor}\label{cor:bound_ext_deriv_vec}
    Under same assumptions as Lemma \ref{lem:bound_secform_boun_proj}, for $v \in \Gamma(TM)$ satisfying
    \begin{equation*}

    \end{equation*}

    \end{equation*}
\end{proof}


\begin{lem}\label{lem:taylo_exp_bounded}
    Let $M$ be a Riemannian submanifold isometrically embedded in $\R^n$ such that the second fundamental form and its derivative are uniformly bounded, i.e.,
    \begin{equation*}
        \sup_{x \in M} \norm{\mathrm{II}_x}_{\op{op}} < \infty,~ \norm{\nabla\mathrm{II}_x}_{\op{op}} < \infty.
    \end{equation*}
    Then its exponential map has
    \begin{equation*}
        \exp_x(u) = x + u + \frac{1}{2}\mathrm{II}_x(u,u) + R_3(x,u),\quad \forall~ x \in M,~u \in T_xM
    \end{equation*}
    where
    \begin{equation*}
        \norm{R_3(x,u)} \leq C \norm{u}^3,
    \end{equation*}
    for some $C > 0$ independent of $x$.
\end{lem}
\begin{rmk}
    Note that for any $X,Y,Z \in \Gamma(TM)$, $\mathrm{II}(X,Y) \in \Gamma(NM)$ and
    \begin{equation*}
        (\nabla_X \mathrm{II})(Y,Z) \defeq \nabla_X^\perp\bc{\mathrm{II}(Y,Z)} - \mathrm{II}(\nabla_XY,Z) - \mathrm{II}(Y,\nabla_XZ),
    \end{equation*}
    where $\nabla_X^\perp N = (\clo{\nabla}_XN)^\perp$ for $N \in \Gamma(TM)$.
\end{rmk}
\begin{proof}
    Fix $x \in M$ and $u \in T_xM$. Let $\gamma \colon [0,1] \sto M$ be the geodesic such that $\gamma(0) = x$ and $\dot{\gamma}(0) = u$. Note that $\gamma \subset \R^n$. Then
    \begin{equation*}
        \gamma^{\prime}(t) = \dot{\gamma}(t),
    \end{equation*}
    and
    \begin{equation*}
        \gamma^{\prime\prime}(t) = \clo{\nabla}_{\gamma^\prime(t)}\gamma^\prime(t) = \nabla_{\dot{\gamma}(t)}\dot{\gamma}(t) + \mathrm{II}(\dot{\gamma}(t),\dot{\gamma}(t)) = \mathrm{II}(\dot{\gamma}(t),\dot{\gamma}(t)).
    \end{equation*}
    Then by the Euclidean Taylor formula for $\gamma(t)$,
    \begin{align*}
        \gamma(1) &= \gamma(0) + \gamma^{\prime}(0) + \frac{1}{2}\gamma^{\prime\prime}(0) + \int_0^1\frac{(1-t)^2}{2} \gamma^{(3)}(t) \dx{t} \\
        &= x + u + \frac{1}{2}\mathrm{II}(u,u) + R_3(x,u),
    \end{align*}
    where
    \begin{equation*}
        R_3(x,u) = \int_0^1\frac{(1-t)^2}{2} \gamma^{(3)}(t) \dx{t}.
    \end{equation*}
    For $\gamma^{(3)}(t)$,
    \begin{equation*}
        \gamma^{(3)}(t) = \clo{\nabla}_{\gamma^\prime(t)}\gamma^{\prime\prime}(t) = \clo{\nabla}_{\dot{\gamma}(t)} \mathrm{II}(\dot{\gamma}(t),\dot{\gamma}(t)).
    \end{equation*}
    Let $X = \dot{\gamma}(t)$ and $S_N(X)$ be the Weingarten map. Then by the Weingarten formula,
    \begin{equation*}
        \clo{\nabla}_{X} \mathrm{II}(X,X) = -S_{\mathrm{\mathrm{II}(X,X)}}(X) + \nabla_X^\perp \mathrm{II}(X,X).
    \end{equation*}
    \begin{itemize}
        \item For the first term, because $\sup_{x \in M}\norm{\mathrm{II}_x}_{\op{op}} \leq C_1 < \infty$, similarly as the proof in Lemma \ref{lem:bound_secform_boun_proj},
        \begin{equation*}
            \norm{S_{\mathrm{\mathrm{II}(X,X)}}(X)} \leq C_1^2\norm{X}^3 = C_1^2\norm{u}^3,
        \end{equation*}
        because $\norm{\dot{\gamma}(t)} \equiv \norm{\dot{\gamma}(0)}$.

        \item For the second term, by the definition
        \begin{align*}
            \nabla_X^{\perp} \mathrm{II}(X, X)&=\left(\nabla_X \mathrm{II}\right)(X, X)+\mathrm{II}\left(\nabla_X X, X\right)+\mathrm{II}\left(X, \nabla_X X\right) \\
            &=\left(\nabla_X \mathrm{II}\right)(X, X),
        \end{align*}
        because $\nabla_XX = 0$ for $X = \dot{\gamma}(t)$. Then by $\norm{\nabla\mathrm{II}_x}_{\op{op}} \leq C_2 < \infty$,
        \begin{equation*}
            \norm{\nabla_X^{\perp} \mathrm{II}(X, X)} \leq C_2 \norm{X}^3 = C_2 \norm{u}^3.
        \end{equation*}
    \end{itemize}
    Therefore,
    \begin{equation*}
        \norm{\gamma^{(3)}(t)} \leq (C_1^2 + C_2)\norm{u}^3.
    \end{equation*}
    It implies that
    \begin{equation*}
        \norm{R_3(x,u)} \leq \frac{C_1^2 + C_2}{6}\norm{u}^3. \qedhere
    \end{equation*}
\end{proof}
